1. 변수와 상수 ★핵심

  • ⭐ 강조 — 변수(variable): 프로그램이 실행되는 동안 값이 바뀌는 정보, 데이터를 저장·처리하는 기억장소
  • ⭐ 강조 — 상수(constant): 한 번 정해지면 프로그램이 끝날 때까지 바뀌지 않는 정보 → const
  • ✍️ 필기 — 기억장소 할당 = 주소(address) = 번지. 컴파일 타임에 변수 이름이 기억장치 주소로 바뀐다 (예: n → 100번지, 메모리 주소 100 101 102 103 …)
  • ✍️ 필기 — 소스(.c) → 컴파일(.obj) → 링크(.exe) → 실행 / 컴파일 타임 ↔ 런 타임
  • 변수 선언: 저장공간 할당 / 변수 초기화: 선언하면서 초깃값 대입
int n;          // 선언: 4byte 공간만 할당 (어떤 값인지 모름)
n = 10;         // 대입
n = 20;         // 변수는 값이 바뀔 수 있다

int value = 0;              // 선언 + 초기화
char name = 'a';            // 'a' 의 ASCII 코드 97 저장
int data1, data2, data3;    // 여러 변수 한 줄 선언
  • 변수 표기법: 카멜 표기법 myName · 헝가리언 표기법 c_name
  • 변수 이름 규칙: 영문자·숫자·밑줄만 사용 / 숫자로 시작 X / 공백 X / 대·소문자 구별 / 예약어(키워드) 사용 X
📌 변수 = 값이 바뀌는 기억장소 · 선언 = 공간 할당 · 초기화 = 선언하면서 값 넣기

예제 ① sample03_01.c — 선언 → 대입 → 값 바꾸기

// sample03_01.c — 변수에 값을 저장하고 출력
#include <stdio.h>
int main() {
    int money;                              // 선언: int(4byte) 저장 공간을 할당
    money = 3000;                           // 대입: 공간에 3000 을 저장
    printf("money에 저장된 값 %d \n", money);

    money = 5000;                           // 변수는 값이 바뀔 수 있다
    printf("money에 저장된 값 %d \n", money);

    int value = 0;                          // 선언 + 초기화를 한 줄에
    int data1, data2, data3;                // 같은 자료형은 쉼표로 여러 개 선언
    data1 = 10; data2 = 20; data3 = 30;
    printf("value=%d data1=%d data2=%d data3=%d\n", value, data1, data2, data3);
    return 0;
}
money에 저장된 값 3000 
money에 저장된 값 5000 
value=0 data1=10 data2=20 data3=30

// ex01_03-var.c — char · int · float 변수 초기화 (실습 1-02 · 1-03)

  • ✍️ 필기 — signed(부호 있음, 기본값) ↔ unsigned(부호 없음) · 예: signed int n; / unsigned int n;
  • ⭐ 강조 — 부호 비트: 맨 앞 1비트가 0이면 양수, 1이면 음수 → 1byte 부호 있음 -128 ~ 127 / 부호 없음 0 ~ 255
  • 10000000 → 부호 있음 -128 / 부호 없음 128 · 11111111 → 부호 있음 -1 / 부호 없음 255
  • ✍️ 필기 정정 — unsigned char (1)~(256) ❌ → 0 ~ 255 ✅ · signed char 는 -128 ~ -1 음수, 0 ~ 127 양수
📌 sizeof 결과는 정수 → 항상 %d 로 출력 · 정수 상수는 int(4) · 실수 상수는 double(8)

예제 — 자료형 · 변수 · 상수의 크기

// 자료형 크기 — sizeof 결과는 정수라 %d 로 출력
#include <stdio.h>
int main() {
    printf("char   : %d byte\n", sizeof(char));     // 문자형
    printf("short  : %d byte\n", sizeof(short));    // 정수형
    printf("int    : %d byte\n", sizeof(int));      // 정수형
    printf("float  : %d byte\n", sizeof(float));    // 실수형
    printf("double : %d byte\n", sizeof(double));   // 실수형
    // long 은 VS(윈도우) 4byte · Mac·Linux 8byte → 강의자료(VS) 기준 4byte

    int n = 10;
    double d = 3.14;
    printf("변수 n : %d byte\n", sizeof(n));        // 변수 이름을 넣어도 된다 → int 크기
    printf("변수 d : %d byte\n", sizeof(d));        // double 크기
    printf("10     : %d byte\n", sizeof(10));       // 정수 상수 = int
    printf("3.14   : %d byte\n", sizeof(3.14));     // 실수 상수 = double
    printf("3.14f  : %d byte\n", sizeof(3.14f));    // f 를 붙이면 float
    char name = 'a';
    printf("변수 name : %d byte\n", sizeof(name));  // char 변수 = 1byte
    return 0;
}

📌 1byte = 8bit · 맨 앞 비트가 0 이면 양수, 1 이면 음수(signed) · unsigned 는 0 이상만

예제 — 같은 비트, 다른 값

// 부호 비트 — 1byte = 8bit, 맨 앞 비트가 0 이면 양수 · 1 이면 음수 (signed)
#include <stdio.h>
int main() {
    signed char   s1 = 127;    // 0111 1111 → signed 최댓값
    signed char   s2 = -128;   // 1000 0000 → signed 최솟값
    unsigned char u1 = 0;      // 0000 0000 → unsigned 최솟값
    unsigned char u2 = 255;    // 1111 1111 → unsigned 최댓값
    printf("signed char   : %d ~ %d\n", s2, s1);
    printf("unsigned char : %d ~ %d\n", u1, u2);

    // 같은 비트라도 signed / unsigned 에 따라 값이 달라진다
    unsigned char u3 = 128;    // 1000 0000 → 부호 없음 : 128
    signed char   s3 = 128;    // 1000 0000 → 부호 있음 : 맨 앞 1 = 음수 → -128
    unsigned char u4 = 255;    // 1111 1111 → 부호 없음 : 255
    signed char   s4 = 255;    // 1111 1111 → 부호 있음 : -1
    printf("1000 0000 → unsigned %d / signed %d\n", u3, s3);
    printf("1111 1111 → unsigned %d / signed %d\n", u4, s4);

    // int n;          ← 기본이 signed (signed int n; 과 같다) : 음수·양수 모두
    // unsigned int n; ← 부호 없는 int : 0 이상만 저장
    return 0;
}
signed char   : -128 ~ 127
unsigned char : 0 ~ 255
1000 0000 → unsigned 128 / signed -128
1111 1111 → unsigned 255 / signed -1

3. 서식 지정자 (% 형식 연산자)

서식의미예시 → 결과
%d10진 정수printf("%d", 100); → 100
%o8진수printf("%o", 100); → 144
%x / %X16진수 (소문자 / 대문자)printf("%x", 100); → 64
%c문자 1개printf("%c", 65); → A
%s문자열 (여러 글자)printf("%s", "안녕하세요");
%f / %lf실수 (float / double)printf("%f", 3.14); → 3.140000
%p주소printf("%p", &age);
  • ✍️ 필기 정정 — 0진법 → %d ❌ → 10진법 %d ✅
  • ⭐ 강조 — 'A' 는 문자 1개(%c), "A" 는 문자열(%s, 끝에 널 문자 포함) → printf("%c", "A"); ❌ · char a1 = "c"; ❌ → 'c' ✅
  • 출력 칸수 조정: printf("[%5d] [%05d] [%-5d]", 7, 7, 7); → [ 7] [00007] [7 ]
📌 % 뒤 글자가 출력 모양을 정한다 · 숫자를 넣으면 칸수 조정

예제 ① 서식 지정자 전부 (%d %o %x %X %c %s %f %lf %ld)

// printf 서식 지정자 — % 뒤의 글자가 "어떤 모양으로 찍을지"를 정한다
#include <stdio.h>
int main() {
    int data = 100;
    printf("10진수 : %d\n", data);        // %d  → 100
    printf("8진수  : %o\n", data);        // %o  → 144   (100 = 1×64 + 4×8 + 4)
    printf("16진수 : %x\n", data);        // %x  → 64    (100 = 6×16 + 4)
    printf("16진수 : %X\n", 255);         // %X  → FF    (대문자 16진수, 소문자 %x 는 ff)
    printf("문자   : %c\n", 'A');         // %c  → 문자 1개
    printf("문자열 : %s\n", "안녕하세요");  // %s  → 여러 글자
    printf("실수   : %f\n", 3.14);        // %f  → 소수점 아래 6자리
    printf("double : %lf\n", 165.7);     // %lf → double 도 6자리
    printf("long   : %ld\n", 123456L);   // %ld → long 정수

    float value = 2.1f;                  // 강의자료 10p
    printf("float  : %f\n", value);      // 2.100000 (6자리)
    return 0;
}

예제 ② 출력 칸수 [%5d] [%05d] [%-5d] (강의자료 11p)

// 출력 칸수 조정 — [ ] 로 칸을 눈으로 확인 (강의자료 11p)
#include <stdio.h>
int main() {
    int data = 7;
    printf("[%d] [%5d]\n", data, data);                   // %5d : 5칸 오른쪽 맞춤
    printf("[%5d] [%05d] [%-5d]\n", data, data, data);    // %05d 빈칸을 0 으로 · %-5d 왼쪽 맞춤
    printf("[%5d] [%5d] [%5d]\n", 7, 77, 777);             // 자릿수가 달라도 오른쪽 끝이 맞는다
    printf("[%-5d] [%-5d] [%-5d]\n", 7, 77, 777);          // 왼쪽 끝이 맞는다
    printf("[%2d]\n", 12345);                               // 칸보다 숫자가 길면 그냥 다 찍는다
    return 0;
}

4. ASCII 코드와 2진수

  • ✍️ 필기 — 문자는 1byte · 'A' = 65(10진수) = 01000001(2진수) · 'a' = 97

  • 64 + 1 = 65 → printf("%c", 65); 는 A 를 출력한다
📌 문자는 1byte 정수(ASCII 코드)로 저장 · 'A' = 65 · 'a' = 97 · 작은따옴표 = 문자, 큰따옴표 = 문자열

예제 — %c 와 %d 로 같은 값을 찍어 보기

// 문자와 ASCII 코드 — 문자는 1byte 정수(ASCII 코드)로 저장된다
#include <stdio.h>
int main() {
    char name = 'a';                        // 'a' 의 ASCII 코드 97 이 저장된다
    printf("%c %d\n", name, name);          // %c → 문자 a · %d → 숫자 97

    printf("65의 ASCII 값은 %c입니다.\n", 65); // 강의자료 10p : %c 는 숫자를 문자로 → A
    printf("'A' 는 %d\n", 'A');              // 65 = 0100 0001
    printf("'a' 는 %d\n", 'a');              // 97

    char c1 = 'A';                          // 작은따옴표 = 문자 1개
    char c2 = 65;                           // 숫자 65 를 넣어도 같은 문자 A
    printf("c1=%c c2=%c\n", c1, c2);

    printf("%s\n", "A");                    // 큰따옴표 = 문자열 → %s
    // ❌ printf("%c", "A");  "A" 는 문자열이라 %c 로 찍으면 안 된다
    // ❌ char a1 = "c";      문자열을 char 에 넣을 수 없다 → 'c' 로
    return 0;
}

5. 상수: const 와 #define

  • ⭐ 강조 — const 상수는 선언과 동시에 초기화해야 하고, 값을 다시 대입하면 컴파일 오류
  • ⭐ 강조 — 매크로 상수 #define: # 으로 시작하는 구문은 컴파일 전에 전처리기가 해석·치환 (끝에 ; 안 붙임)
// ex01_04-const.c: 상수
#include <stdio.h>
int main() {
    const int a1 = 1000;    // 상수는 선언과 동시에 초기화해야
    printf("a1: %d \n", a1);
    // a1 = 2000;           // error: 상수는 재할당 불가능
    printf("a1: %d \n", a1);
    return 0;
}
// #define 매크로 상수 + const 상수
#define DATA1 100
#include <stdio.h>
int main() {
    const int a1 = 1000;
    printf("a1 %d DATA: %d\n", a1, DATA1);    // a1 1000 DATA: 100
    return 0;
}
📌 const = 선언과 동시에 초기화, 재대입하면 컴파일 오류 · #define = 전처리기가 컴파일 전에 값으로 바꿔 놓음 (끝에 ; 없음)

예제 ① const 없이 — 값이 바뀐다

// ex01_04-const.c ① const 없이 — 변수는 다시 대입할 수 있다
#include <stdio.h>
int main() {
    int a1 = 1000;
    printf("a1: %d \n", a1);    // 1000
    a1 = 2000;                  // 변수라서 값이 바뀐다
    printf("a1: %d \n", a1);    // 2000
    return 0;
}
a1: 1000 
a1: 2000

예제 ② const 를 붙이면 — 바꿀 수 없다

// ex01_04-const.c ② const 를 붙이면 상수 → 다시 대입하면 컴파일 오류
#include <stdio.h>
int main() {
    const int a1 = 1000;        // 상수는 선언과 동시에 초기화해야 한다
    printf("a1: %d \n", a1);
    // a1 = 2000;               // ❌ error: 상수에 값 재할당 → 컴파일되지 않는다
    printf("a1: %d \n", a1);    // 그대로 1000
    return 0;
}

// 매크로 상수 #define — 컴파일 전에 전처리기가 이름을 값으로 바꿔 놓는다 (강의자료 13p)
#include <stdio.h>
#define Even 2                        // 끝에 ; 를 붙이지 않는다
#define MSG "==================="     // 문자열도 정의할 수 있다
#define DATA1 100
int main() {
    // ⚠️ 강의자료 13p 는 printf("Even: %d\n, Even"); → 따옴표 위치가 틀려 값이 안 찍힌다
    printf("Even: %d\n", Even);         // ✅ Even 이 2 로 바뀌어 컴파일된다
    printf("Message: %s\n", MSG);       // MSG → "===================" 로 바뀐다

    const int a1 = 1000;                // const 상수와 같이 쓰기
    printf("a1 %d DATA: %d\n", a1, DATA1);
    printf("Even * DATA1 = %d\n", Even * DATA1);   // 2 * 100 으로 바뀌어 계산된다
    return 0;
}

Message: ===================
a1 1000 DATA: 100
Even * DATA1 = 200

6. scanf() 와 & 주소 연산자

  • ⭐ 강조 — scanf("%d", &age); : & 로 변수의 시작 주소를 넘겨야 입력값이 그 기억장소에 저장된다
  • printf("%p, %d\n", &age, age); → 주소와 값을 출력 (예: 0x16f3eb3d8, 21)
  • VS 에서 scanf 를 쓰려면 맨 위에 #define _CRT_SECURE_NO_WARNINGS · 앞의 밑줄이나 철자가 틀리면 C4996 오류
// Prj06/06scanf.c  printf(), scanf()
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
int main(void) {
    int month = 0;
    printf("1년은 몇 달? ");
    scanf("%d", &month);
    printf("1년은 %d달\n\n", month);

    int snum, credit;
    printf("당신의 학번과 신청 학점은? ");
    scanf("%d %d", &snum, &credit);
    printf("학번:  %d 신청학점: %d\n", snum, credit);
    return 0;
}
📌 & = 변수의 시작 주소 → scanf 는 주소를 받아야 그 공간에 입력값을 저장 · VS 는 맨 위에 #define _CRT_SECURE_NO_WARNINGS

예제 ① Prj06/06scanf.c (실습 2-01) — 입력: 12 / 20232629 18

// Prj06/06scanf.c — scanf 로 키보드 입력 받기 (실습 2-01)
#define _CRT_SECURE_NO_WARNINGS    // VS 에서 scanf 를 쓰려면 맨 위에 (없으면 C4996 오류)
#include <stdio.h>
int main(void) {
    int month = 0;
    printf("1년은 몇 달? ");
    scanf("%d", &month);           // & = 변수의 주소 → 입력값을 month 공간에 저장
    printf("1년은 %d달\n\n", month);

    int snum, credit;
    printf("당신의 학번과 신청 학점은? ");
    scanf("%d %d", &snum, &credit);  // 공백(스페이스·엔터)으로 구분해서 두 개 입력
    printf("학번:  %d 신청학점: %d\n", snum, credit);
    return 0;
}

당신의 학번과 신청 학점은? 20232629 18

예제 ② 자료형에 맞는 scanf 서식 — 입력: 21 165.7

// scanf 서식 — 받을 변수의 자료형에 맞춘다 (double 은 반드시 %lf)
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
int main(void) {
    char gender = 'w';
    int age;
    double height;
    printf("나이와 신장은? ");
    scanf("%d %lf", &age, &height);      // int → %d · double → %lf
    printf("성별: %c, 나이: %d, 신장: %lf\n", gender, age, height);
    // ⚠️ printf 는 double 에 %f 도 되지만, scanf 는 %lf 가 아니면 값이 제대로 안 들어간다
    return 0;
}
나이와 신장은? 21 165.7
성별: w, 나이: 21, 신장: 165.700000

예제 ③ %p 로 주소 찍어 보기

// & 주소 연산자 — 변수가 메모리 몇 번지에 있는지 (%p = 주소 출력)
#include <stdio.h>
int main(void) {
    int age = 25;
    printf("%p, %d\n", &age, age);     // 주소, 값 (강의자료 12p)

    int a = 1, b = 2;
    printf("a 의 주소 %p\n", &a);        // 변수마다 주소가 다르다
    printf("b 의 주소 %p\n", &b);
    // 주소 값은 실행할 때마다 · 컴퓨터마다 다르다 (VS 는 00000012345ABCDE 모양)
    // scanf("%d", age);  ❌ & 를 빠뜨리면 저장할 주소를 모른다 → 실행 중 오류
    return 0;
}
0x16dda6938, 25
a 의 주소 0x16dda6934
b 의 주소 0x16dda6930
주소 값은 실행할 때마다 · 컴퓨터마다 다릅니다(위는 Mac 에서 실행한 값). 중요한 건 &변수 = 그 변수가 있는 메모리 번지 라는 것.

8. ★ 시험문제 정리

① 강의자료 23p — [실습 1-05] ex01_05-sizeof.c 실행결과 쓰기

// ex01_05-sizeof.c
#include <stdio.h>
int main() {
    char gender;
    int age;
    float weight;
    double height;

    printf("변수의 할당크기\n");
    printf("gender: %d, age: %d, weight: %d, height: %d\n\n",
        sizeof(gender), sizeof(age), sizeof(weight), sizeof(height));

    printf("자료형 크기\n");
    printf("char: %d, int: %d, float: %d, double: %d\n\n",
        sizeof(char), sizeof(int), sizeof(float), sizeof(double));

    printf("정수형상수: %d, 실수형상수: %d\n",
        sizeof(10), sizeof(3.14159));
    return 0;
}
변수의 할당크기
gender: 1, age: 4, weight: 4, height: 8

자료형 크기
char: 1, int: 4, float: 4, double: 8

정수형상수: 4, 실수형상수: 8
  • ⭐ 강조 — sizeof 는 크기(바이트 수)를 정수로 돌려준다 → %d 로 출력
  • ⭐ 강조 — sizeof(10) → 정수 상수는 int → 4 / sizeof(3.14159) → 실수 상수는 double → 8 (3.14f 처럼 f 를 붙이면 float → 4)
  • ✍️ 필기 — (수업 중 "시험문제!") sizeof(float) 을 %f, sizeof(double) 을 %lf, sizeof(char) 를 %c 로 찍으면 ❌ → 전부 %d → int: 4 · float: 4 · double: 8 · char: 1

9. 오류 수정 모음 ★빈칸채우기·오류수정 대비

  • printf("문자 리터럴 %c\n", "A"); ❌ → 'A' ✅
  • a1 = "c"; ❌ → a1 = 'c'; ✅
  • 서식표 빈칸: short → %s ❌ → %d ✅ · long → % ❌ → %ld ✅ · 0진법 ❌ → 10진법 %d ✅
  • unsigned char (1)~(256) ❌ → 0 ~ 255 ✅
  • #define CRT_SECUE_NO_WARNINGS ❌ → #define _CRT_SECURE_NO_WARNINGS ✅
  • printf("float: %f \n", sizeof(float)); ❌ → %d ✅ (sizeof 결과는 정수)
✍️ 수업 중 원본 필기 (정리 전 메모) 
변수(variable): 기억장소 · 상수(constant): const · 기억장소 할당: 주소(address). 번지 · int n; n=10; n=20;
char a1; 1byte (-128)~(+127) · int a2; 4byte · float a3; 4byte · double a4; 8byte
signed char (-128)~(+127) · unsigned char (1)~(256) · -128~-1:음수 · 0~255:양수
자료형(data types): 부호가 있는 문자형 signed int n;(default 값) · 부호가 없는 문자형 unsigned int n;
0진법 → %d · 8진법 → %o · 16진법 → %x
문자형 char → 1바이트 → %c · 숫자형 short → 2바이트 → %d · int → 4바이트 → %d · long → 8바이트 → %ld · 실수형 float → 4바이트 → %f · double → 8바이트 → %lf
