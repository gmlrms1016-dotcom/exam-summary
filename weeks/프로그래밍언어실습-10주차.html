📄 왼쪽 — 배열 요소마다 printf 를 한 줄씩

#include <stdio.h>                                  // printf 를 쓰기 위한 표준 입출력 헤더

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int array1[3];                                  // int 3칸 배열 → array1[0] · array1[1] · array1[2]
    array1[0] = 10;                                 // 0번 칸에 10
    array1[1] = 20;                                 // 1번 칸에 20
    array1[2] = 30;                                 // 2번 칸에 30

    printf("array1[0]: %d\n", array1[0]);           // array1[0] = 10 → "array1[0]: 10"
    printf("array1[1]: %d\n", array1[1]);           // array1[1] = 20 → "array1[1]: 20"
    printf("array1[2]: %d\n", array1[2]);           // array1[2] = 30 → "array1[2]: 30"

    return 0;                                       // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                   // main 끝

📄 오른쪽 — 같은 출력을 for 문 하나로

#include <stdio.h>                                  // printf 를 쓰기 위한 표준 입출력 헤더

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int array1[3];                                  // int 3칸 배열
    int i;                                          // ✅ 반복에 쓸 첨자 변수 — 왼쪽에는 없던 줄
    array1[0] = 10;                                 // 0번 칸에 10
    array1[1] = 20;                                 // 1번 칸에 20
    array1[2] = 30;                                 // 2번 칸에 30

    for (i = 0; i < 3; i++)                         // i = 0 → 1 → 2 (3번 · i < 3 = 칸 수) — ✍️ "i 는 0부터 2까지 반복"
        printf("array1[%d]: %d\n", i, array1[i]);   // %d 두 개 ← i 와 array1[i] → "array1[0]: 10" · "array1[1]: 20" · "array1[2]: 30"

    return 0;                                       // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                   // main 끝

② 🔥 기말고사 — 14쪽 sizeof 문제 ([실습 1-01] array01.c)

#include <stdio.h>                                  // printf 를 쓰기 위한 표준 입출력 헤더

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int count[5];   // 정수형 1차원 배열            // int(4바이트) 5칸 → 4 × 5 = 20바이트
    int i;                                          // 반복에 쓸 첨자

    printf("배열의 전체크기:%d, 하나의 원소크기: %d\n",   // %d 두 개
        sizeof(count), sizeof(count[0]));           // sizeof(배열 이름) = 전체 20 · sizeof(count[0]) = 원소 하나 4

    count[0] = 100;   // 각각의 원소에 값을 할당    // 0번 칸 = 100
    count[1] = 200;                                 // 1번 칸 = 200
    count[2] = 300;                                 // 2번 칸 = 300
    count[3] = 400;                                 // 3번 칸 = 400
    count[4] = 500;                                 // 4번 칸 = 500 (마지막 칸 = 칸 수 - 1)

    printf("\n%d, %d, %d, %d, %d \n",               // 맨 앞 \n → 빈 줄 하나 먼저 · 끝 %d 뒤에 칸 하나
        count[0], count[1], count[2], count[3], count[4]);   // 100, 200, 300, 400, 500

    for (i = 0; i < 5; i++)                         // i = 0 ~ 4 (5번)
        printf("%d, ", count[i]);                   // "100, " → "200, " → … → "500, " (한 줄로 이어짐)
    printf("\n");                                   // 다 찍고 줄바꿈

    return 0;                                       // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                   // main 끝
배열의 전체크기:20, 하나의 원소크기: 4

  • 🔥 칸 수 = sizeof(count) / sizeof(count[0]) = 20 / 4 = 5 (📄 과제 07arysize.c 와 같은 방법)
  • 둘째 줄이 빈 줄 — 두 번째 printf 서식이 "\n%d, …" 로 \n 부터 시작해서 (시험 ✍️ “\n 있나 없나 주의”)
  • 🤖 sizeof 의 결과는 size_t(부호 없는 정수)라 정확한 서식은 %zu · 수업 코드처럼 %d 로 찍어도 VS 에서 같은 값이 나와요 (경고만)
  • ✍️ 필기 — int count[5]; → int = 4byte 이므로 20byte 필요 · short student[10]; → short = 2byte × 10 = 20byte

1. ✍️ [이론] 배열

  • ✍️ 배열 = 동일한 자료형을 갖는 기억장소를 연속된 공간에 할당한 데이터 모임 (📄 4쪽: 하나의 이름으로 관리 · 0부터 시작하는 인덱스번호)
  • ✍️ 첨자(인덱스) 는 0부터 — int count[5]; → count[0] ~ count[4] (📄 4쪽: 선언한 개수의 범위를 넘으면 오류가 날 수 있다)
  • 📄 배열이름 = 배열의 시작주소 — count == &count[0] (5쪽) · 원소값 = 배열변수[첨자] · 원소주소 = &배열변수[첨자] (15쪽)
✍️ 차원선언 예 (필기)뜻
1차원int array[5]학생 한 명의 5과목
2차원int array[10][5]10명의 학생의 5과목
3차원int array[3][10][5]3반의 각각 10명의 학생의 5과목
  • ✍️ 컴파일 시간(compile time) / 실행 시간(run time) — 🤖 배열 크기는 컴파일 시간에 정해져야 해서 int a[5]; 처럼 상수로 써요 (#define SIZE 5 도 상수)

2. ✍️ 배열 선언 · 초기화

  • ✍️ 변수 초기화 = int x; x=10; · 배열 선언 = int count[5]; · 배열 초기화 = int count[5] = {10, 20, 30, 40, 50};
  • ⚠️ 정정 — 필기 int count[5]={10.20.30.40.50};; → 값 사이는 쉼표(,) · 세미콜론은 하나. 10.20.30 처럼 점을 쓰면 실수 10.2 처럼 읽혀서 컴파일 오류예요

✍️ 수업 파일 array4.c — for 문으로 하나씩 대입

#include<stdio.h>                                   // printf (< 앞 공백은 없어도 됨)
int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int x[5] = { 1,2,3,4,5 };                       // 원본 배열 x
    int y[5];                                       // 복사받을 배열 y (아직 쓰레기값)
    int i;                                          // 첨자

    for (i = 0; i < 5; i++) {                       // i = 0 ~ 4
        y[i] = x[i];                                // y[0] = 1 → y[1] = 2 → … → y[4] = 5 (원소 하나씩)
    }                                               // for 끝 → y 가 x 와 같아짐
    printf("배열 출력\n");                          // 제목 줄
    for (i = 0; i < 5; i++) {                       // i = 0 ~ 4
        printf("%d,", y[i]);                        // "1," "2," … "5," (쉼표 뒤 칸 없음)
    }                                               // for 끝
    printf("\n=====================\n");           // 줄바꿈 → 구분선 → 줄바꿈

    return 0;                                       // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                   // main 끝

4. 배열과 주소 · 2차원 배열 (15 · 16쪽 · ✍️ array06)

📄 15쪽 [실습 1-02] array02.c — 원소값과 원소주소 (%p)

#include <stdio.h>                                  // printf 를 쓰기 위한 표준 입출력 헤더

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int  count[5] = { 10,20,30,40,50 };   //배열 초기화   // 5칸에 10 ~ 50
    int i;                                          // 첨자

    for (i = 0; i < 5; i++)                         // i = 0 ~ 4
        printf("%d, %p \n", count[i], &count[i]);   // 값 · 주소(&) → 주소는 int 크기만큼 4씩 커짐

    return 0;                                       // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                   // main 끝

📄 16쪽 [실습 1-06] array06.c (✍️ 수업에서 직접 침) — 2차원 배열의 크기와 주소

#include <stdio.h>                                  // printf 를 쓰기 위한 표준 입출력 헤더

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
	int i, j;                                       // 행 첨자 i · 열 첨자 j
	int count[2][3] = { 10,20,30,40,50,60 };        // 2행 3열 → 0행 = 10 20 30 · 1행 = 40 50 60 (행부터 차례로 채움)

	printf("배열 전체의 크기: %d, 한 행의 크기 : %d, 하나의 원소의 크기: %d \n\n",   // 끝에 \n 두 번 → 빈 줄
		sizeof(count), sizeof(count[0]), sizeof(count[0][0]));   // 6칸 × 4 = 24 · 한 행 3칸 × 4 = 12 · 원소 4

	for (i = 0; i < 2; i++) {                       // 행 i = 0, 1
		for (j = 0; j < 3; j++)                     // 열 j = 0, 1, 2
			printf("%d, ", count[i][j]);            // 0행: "10, 20, 30, " · 1행: "40, 50, 60, "
		printf("\n");                               // 한 행을 다 찍으면 줄바꿈
	}                                               // 바깥 for 끝

	printf("\n배열의 시작주소: %p\n, 0행의 시작주소: %p\n, 1행의시작주소: %p\n",   // 원본 서식 그대로 (쉼표가 다음 줄 맨 앞)
		count, count[0], count[1]);                 // count == count[0] (같은 주소) · count[1] = 12바이트 뒤

	return 0;                                       // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                   // main 끝

5. 실습 — 입력받아 배열에 저장 (✍️ ex01_00-array3.c)

// [실습 1-03] 배열                                  // 정수 5개를 입력받아 배열에 저장 → 출력

    int count[5];  //배열 선언                      // 5칸 (아직 쓰레기값)

    printf("\n");                                   // 줄바꿈

6. ✍️ 수업 퀴즈 ① 3 × 3 지그재그로 채우기 (quiz01-01-array1.c)

// quiz01-01-array1.c                               // 1 ~ 9 를 → ← → 방향으로 번갈아 채우기
#include <stdio.h>                                  // printf 를 쓰기 위한 표준 입출력 헤더

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
	int a[3][3];                                    // 3행 3열
	int j = 0, sw = 1, k = 1;                       // j = 열 위치 · sw = 방향(+1 →, -1 ←) · k = 넣을 숫자 1부터

	for (int i = 0; i < 3; i++) {                   // 행 i = 0, 1, 2
		while (j <= 2 && j >= 0) {                  // 열이 0 ~ 2 안에 있는 동안
			a[i][j] = k++;                          // 지금 칸에 k 를 넣고 k 1 증가 (0행: 1 2 3)
			j += sw;                                // 방향대로 한 칸 이동 → 0행 끝나면 j = 3 (밖)
		}                                           // while 끝 → j 가 배열 밖으로 나감
		sw *= -1;                                   // 방향 바꾸기: 1 → -1 (←) → 1 (→)
		j += sw;                                    // 밖으로 나간 j 를 한 칸 되돌림 → 3 → 2 (1행은 2부터 ←)
	}                                               // 1행: 4 5 6 을 j = 2 · 1 · 0 에 → 6 5 4 · 2행: 7 8 9

	for (int i = 0; i < 3; i++) {                   // 출력: 행 i
		for (int j = 0; j < 3; j++)                 // 열 j (위의 j 와 다른 새 지역변수)
			printf("%3d ", a[i][j]);                // 3칸 오른쪽 정렬 + 칸 하나
		printf("\n");                               // 행마다 줄바꿈
	}                                               // 바깥 for 끝

	return 0;                                       // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                   // main 끝

7. ✍️ 수업 퀴즈 ② 5 × 5 뱀 모양 출력 (퀴즈.c)

✍️ 수업에서 친 코드 — 값을 직접 초기화해서 출력

#define _CRT_SECURE_NO_WARNINGS                     // Visual Studio 에서 scanf 보안 경고 끄기 (이 코드는 scanf 가 없어서 없어도 됨)
#include <stdio.h>                                  // printf 를 쓰기 위한 표준 입출력 헤더

int main(void)                                      // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
{                                                   // main 시작
	int arr[5][5] = {                               // 5행 5열을 행마다 중괄호로 초기화
		{1,2,3,4,5},                                // 0행 → 1 ~ 5
		{10, 9, 8, 7, 6 },                          // 1행 ← 6 ~ 10 을 거꾸로
		{11, 12, 13, 14, 15 },                      // 2행 → 11 ~ 15
		{20,19,18,17,16},                           // 3행 ← 16 ~ 20 을 거꾸로
		{ 21,22,23,24,25 }                          // 4행 → 21 ~ 25
	};                                              // 초기화 끝

	for (int i = 0; i < 5; i++) {                   // 행 i = 0 ~ 4
		for (int j = 0; j < 5; j++) {               // 열 j = 0 ~ 4
			printf("%5d",arr[i][j]);                // 5칸 오른쪽 정렬
		}                                           // 안쪽 for 끝
		printf("\n");                               // 행마다 줄바꿈
	}                                               // 바깥 for 끝
	return 0;                                       // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                   // main 끝

🤖 값을 손으로 쓰지 않고 반복문으로 채우는 방법 (퀴즈 ① 과 같은 생각 · 홀수 행만 거꾸로)

#include <stdio.h>                                  // printf 를 쓰기 위한 표준 입출력 헤더

int main(void) {   // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
	int arr[5][5];                                  // 5행 5열 (반복문으로 채움)
	int k = 1;                                      // 넣을 숫자 1부터

	for (int i = 0; i < 5; i++) {                   // 행 i = 0 ~ 4
		for (int j = 0; j < 5; j++) {               // 넣는 차례 j = 0 ~ 4
			if (i % 2 == 0)                         // 짝수 행(0 · 2 · 4) → 왼쪽부터
				arr[i][j] = k++;                    // 0행: arr[0][0]=1 … arr[0][4]=5
			else                                    // 홀수 행(1 · 3) → 오른쪽부터
				arr[i][4 - j] = k++;                // 1행: arr[1][4]=6 … arr[1][0]=10
		}                                           // 안쪽 for 끝
	}                                               // 바깥 for 끝

	for (int i = 0; i < 5; i++) {                   // 출력: 행 i
		for (int j = 0; j < 5; j++)                 // 열 j
			printf("%5d", arr[i][j]);               // 5칸 오른쪽 정렬
		printf("\n");                               // 행마다 줄바꿈
	}                                               // 바깥 for 끝
	return 0;                                       // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                   // main 끝
