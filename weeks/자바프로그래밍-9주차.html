1. 복습 — 생성자 3개 · this() (Book · BookMain)

// Book.java + BookMain.java 를 한 파일로 — 생성자 3개 · this() (4장 예제 4-5)
class Book {                                    // 책 설계도 · public 은 파일 이름과 같은 클래스(연습)에만
    String title;                               // 필드: 제목
    String author;                              // 필드: 저자

    Book() {                                    // ① 기본 생성자 — 매개변수 0개
        this("", "");                           // 첫 문장 this(…) → ③ 을 먼저 불러 title · author ← 빈 문자열 ""
        System.out.println("기본 생성자 호출됨");    // ③ 이 끝나고 돌아와서 출력 → 객체를 만들 때 찍힘
    }                                           // ① 끝

    Book(String title) {                        // ② 매개변수 1개 — 제목만 받음
        this(title, "작자미상");                  // 저자는 "작자미상" 으로 채워서 ③ 을 부름
    }                                           // ② 끝

    Book(String title, String author) {         // ③ 매개변수 2개 — 실제로 필드에 넣는 생성자
        this.title = title;                     // this.title(필드) ← 매개변수 title
        this.author = author;                   // this.author(필드) ← 매개변수 author
    }                                           // ③ 끝

    void printInfo() {                          // 메소드 — 필드 두 개를 한 줄로 출력
        System.out.println("제목: " + title + ", 저자: " + author);   // "제목: " + 제목 + ", 저자: " + 저자
    }                                           // printInfo 끝
}                                               // Book 끝

public class 연습 {                              // 원래 이름 BookMain → 파일 이름 연습.java
    public static void main(String[] args) {    // 프로그램 시작
        Book littlePrince = new Book("어린왕자", "생텍쥐페리");   // 매개변수 2개 → ③ 바로 → 출력 없음
        Book loveStory = new Book("춘향전");                   // 1개 → ② → this(…) → ③ → 저자 "작자미상"
        Book empty = new Book();                              // 0개 → ① → this("", "") → ③ → 돌아와서 "기본 생성자 호출됨" 출력

        littlePrince.printInfo();               // 제목: 어린왕자, 저자: 생텍쥐페리
        loveStory.printInfo();                  // 제목: 춘향전, 저자: 작자미상
        empty.printInfo();                      // 빈 문자열이라 "제목: , 저자: " 만 보임
    }                                           // main 끝
}                                               // 연습 끝

제목: 춘향전, 저자: 작자미상

this(…)같은 클래스의 다른 생성자를 부름 (생성자 코드를 한 곳에 모음)
어디서생성자 안에서만
언제반드시 생성자 코드의 첫 문장 — println 다음에 쓰면 call to this must be first statement in constructor (javac 로 직접 확인)
this. 와 차이this.title = 이 객체의 필드 · this(…) = 다른 생성자 호출
출력 첫 줄이 기본 생성자 호출됨 인 이유 — 세 번째 객체 new Book() 을 만드는 순간 ① 생성자가 실행돼서 printInfo 세 줄보다 먼저 찍혀요 · 마지막 줄은 제목 · 저자가 빈 문자열("")이라 제목: , 저자: 만 보여요.
📄 교재 예제 4-5 는 생성자 호출됨 과 show()(제목 · 저자를 띄어쓰기로 출력)인데, 수업 파일은 기본 생성자 호출됨 과 printInfo()(제목: … , 저자: …) 로 바꿨어요 — 위 코드는 수업 파일대로예요.

2. 메소드 오버로딩 (MethodSample · MethodMain)

// MethodSample.java + MethodMain.java 를 한 파일로 — 메소드 오버로딩 (4장 42 · 43쪽)
class MethodSample {                            // getSum 이 3개 — 이름은 같고 매개변수가 다름
    public int getSum(int i, int j) {           // ① int 2개
        return i + j;                           // 두 수의 합을 int 로 돌려줌
    }                                           // ① 끝
    public int getSum(int i, int j, int k) {    // ② int 3개 — 개수가 달라서 오버로딩 OK
        return i + j + k;                       // 세 수의 합
    }                                           // ② 끝
    public double getSum(double i, double j) {  // ③ double 2개 — 타입이 달라서 오버로딩 OK
        return i + j;                           // 실수 합을 double 로
    }                                           // ③ 끝
}                                               // MethodSample 끝

public class 연습 {                              // 원래 이름 MethodMain
    public static void main(String[] args) {    // 프로그램 시작
        MethodSample a = new MethodSample();    // 메소드를 부를 객체 a

        int i = a.getSum(1, 2);                 // int 2개 → ① → 1 + 2 → 3
        int j = a.getSum(1, 2, 3);              // int 3개 → ② → 1 + 2 + 3 → 6
        double k = a.getSum(1.1, 2.2);          // double 2개 → ③ → 1.1 + 2.2 → 3.3000000000000003 (실수 오차)

        System.out.println(i);                  // 3
        System.out.println(j);                  // 6
        System.out.println(k);                  // 3.3000000000000003
    }                                           // main 끝
}                                               // 연습 끝

오버로딩 (4장 42 · 43쪽)뜻
성공 조건이름이 같고 매개변수의 개수나 타입이 다름
리턴 타입상관없음 — int getSum(int i, int j) 와 double getSum(int i, int j) 는 매개변수가 같아서 ❌ 실패 → method getSum(int,int) is already defined (javac 로 확인)
호출넣은 값의 개수 · 타입에 맞는 메소드가 골라져요 — getSum(1, 2) · getSum(1, 2, 3) · getSum(1.1, 2.2)
🤖 1.1 + 2.2 가 3.3 이 아니라 3.3000000000000003 — double 은 2진수로 저장해서 1.1 · 2.2 를 딱 맞게 담지 못하는 실수 오차 때문이에요 (실행한 값 그대로).

3. 복습 — for-each (ForeachEx1 · ForeachEx2)

// ForeachEx1.java + ForeachEx2.java 를 한 파일로 — for-each (향상된 for 문)
public class 연습 {                              // 원래 이름 ForeachEx1 · ForeachEx2
    public static void main(String[] args) {    // 프로그램 시작
        String[] fruits = {"사과", "바나나", "체리"};   // 문자열 배열 3칸
        for (String fruit : fruits) {           // 배열 칸을 처음부터 하나씩 fruit 에 꺼냄 (인덱스 없음)
            System.out.println(fruit);          // 사과 → 바나나 → 체리
        }                                       // for-each 끝

        int pointsArr[] = {65, 82, 95, 78, 40, 88};       // 점수 6개
        System.out.println("합격자 목록 80점 이상");          // 제목 줄
        for (int point : pointsArr) {           // 65, 82, 95, 78, 40, 88 이 차례로 point 에
            if (point >= 80) {                  // 80 이상만 → 82 · 95 · 88
                System.out.println(point + "점");   // 82 + "점" → 82점 (숫자 + 문자열 → 문자열)
            }                                   // if 끝
        }                                       // for-each 끝
    }                                           // main 끝
}                                               // 연습 끝

// Sample.java + AccessEx.java 를 한 파일로 (4장 예제 4-10) — ❌ 컴파일 오류가 나는 원래 코드
class Sample {                                  // 같은 패키지(같은 폴더)의 클래스
    public int a;                               // public → 어디서나 접근 OK
    private int b;                              // private → Sample 클래스 안에서만
    int c;                                      // 디폴트(아무것도 안 씀) → 같은 패키지 OK
}                                               // Sample 끝

public class 연습 {                              // 원래 이름 AccessEx
    public static void main(String[] args) {    // 프로그램 시작
        Sample aClass = new Sample();           // Sample 객체
        aClass.a = 10;                          // ✅ public
        aClass.b = 10;                          // ❌ private 을 다른 클래스(연습)에서 → 컴파일 오류
        aClass.c = 10;                          // ✅ 디폴트 · 같은 패키지
    }                                           // main 끝
}                                               // 연습 끝

// 🤖 고친 코드 — private b 는 Sample 안의 메소드로만 바꾸고 읽기
class Sample {                                  // 같은 패키지의 클래스
    public int a;                               // public
    private int b;                              // private → 이 클래스 안에서만
    int c;                                      // 디폴트
    void setB(int b) {                          // Sample 안의 메소드 → private b 에 접근 가능
        this.b = b;                             // 필드 b ← 매개변수 b
    }                                           // setB 끝
    int getB() {                                // b 값을 돌려주는 메소드
        return b;                               // 클래스 안이라 OK
    }                                           // getB 끝
}                                               // Sample 끝

public class 연습 {                              // 파일 이름 연습.java
    public static void main(String[] args) {    // 프로그램 시작
        Sample aClass = new Sample();           // Sample 객체
        aClass.a = 10;                          // public → 바로 대입
        aClass.setB(20);                        // private → 메소드를 거쳐서 20
        aClass.c = 30;                          // 디폴트 → 같은 패키지라 바로 대입
        System.out.println(aClass.a + " " + aClass.getB() + " " + aClass.c);   // 10 20 30
    }                                           // main 끝
}                                               // 연습 끝

5. ✍️ static — 클래스에 하나 · 모든 객체가 공유 (Counter · StaticEx1)

(4장 57 · 58쪽)non-static 멤버static 멤버
선언int n; · void g() {…}static int m; · static void f() {…}
공간적 특성멤버는 객체마다 별도 존재 · 인스턴스 멤버멤버는 클래스당 하나 · 객체 밖 별도 공간 · 클래스 멤버
시간적 특성객체 생성 시 생김 · 객체가 사라지면 같이 사라짐클래스 로딩 시 생김 · 객체가 생기기 전에도 사용 가능 · 프로그램이 끝날 때 사라짐
공유의 특성공유되지 않음 (객체 안에 각각)같은 클래스의 모든 객체가 공유
4장 56쪽 그림 — 눈은 사람마다 따로(non-static) · 공기는 모두가 함께(static)
// Counter.java + StaticEx1.java 를 한 파일로 — static 필드 = 모든 객체가 공유
class Counter {                                 // 카운터 설계도
    int instanceCount;                          // non-static(인스턴스) 필드 — 객체마다 따로 · 기본값 0
    static int gongyuCount;                     // static(클래스) 필드 — 클래스에 하나 · 모든 객체가 공유 · 기본값 0
    void add1() {                               // 둘 다 1씩 늘리는 메소드
        instanceCount++;                        // 이 객체의 칸만 +1
        gongyuCount++;                          // 공유 칸 +1 → 다른 객체가 늘린 값에 이어서
    }                                           // add1 끝
}                                               // Counter 끝

public class 연습 {                              // 원래 이름 StaticEx1
    public static void main(String[] args) {    // 프로그램 시작
        Counter cnt1 = new Counter();           // 객체 1 — instanceCount 0
        Counter cnt2 = new Counter();           // 객체 2 — instanceCount 0 (cnt1 과 따로)
        cnt1.add1();                            // cnt1.instanceCount 0 → 1 · gongyuCount 0 → 1
        cnt2.add1();                            // cnt2.instanceCount 0 → 1 · gongyuCount 1 → 2 (같은 칸)
        System.out.println("공유 값 클래스 static = " + Counter.gongyuCount);   // 클래스 이름으로 → 2
        System.out.println("객체 값1 = " + cnt1.instanceCount);              // cnt1 의 칸 → 1
        System.out.println("객체 값2 = " + cnt2.instanceCount);              // cnt2 의 칸 → 1
        System.out.println("공유 값 객체 static = " + cnt1.gongyuCount);     // 객체 이름으로 읽어도 같은 칸 → 2
    }                                           // main 끝
}                                               // 연습 끝
공유 값 클래스 static = 2
객체 값1 = 1
객체 값2 = 1
공유 값 객체 static = 2
cnt1 · cnt2 가 add1() 을 한 번씩 → 각자의 instanceCount 는 1 · 공유 칸 gongyuCount 는 2 · static 은 Counter.gongyuCount (클래스 이름)로도, cnt1.gongyuCount (객체 이름)로도 읽을 수 있고 같은 칸이에요.

6. static 메소드 — Calc (예제 4-11)

// Calc.java + CalcEx.java 를 한 파일로 — static 메소드 (4장 예제 4-11)
class Calc {                                    // 계산 도구 모음 (객체 없이 쓰는 전역 함수처럼)
    public static int abs(int a) {              // 절댓값 — static 이라 Calc.abs(…) 로 바로 호출
        return a > 0 ? a : -a;                  // 삼항: -5 > 0 거짓 → -a → -(-5) → 5
    }                                           // abs 끝
    public static int max(int a, int b) {       // 큰 값
        return a > b ? a : b;                   // 10 > 8 참 → a → 10
    }                                           // max 끝
    public static int min(int a, int b) {       // 작은 값
        return a > b ? b : a;                   // -3 > -8 참 → b → -8
    }                                           // min 끝
}                                               // Calc 끝

public class 연습 {                              // 원래 이름 CalcEx
    public static void main(String[] args) {    // 프로그램 시작
        System.out.println(Calc.abs(-5));       // new 없이 클래스 이름으로 → 5
        System.out.println(Calc.max(10, 8));    // 10
        System.out.println(Calc.min(-3, -8));   // -8
    }                                           // main 끝
}                                               // 연습 끝

7. 객체 배열 (BookArray · 예제 4-7)

// Book.java + BookArray.java 를 한 파일로 — 객체 배열 (4장 예제 4-7)
import java.util.Scanner;                       // 키보드 입력 도구

class Book {                                    // 책 설계도 (이 예제에 필요한 생성자만)
    String title;                               // 필드: 제목
    String author;                              // 필드: 저자
    Book(String title, String author) {         // 매개변수 2개 생성자
        this.title = title;                     // 필드 ← 입력한 제목
        this.author = author;                   // 필드 ← 입력한 저자
    }                                           // 생성자 끝
}                                               // Book 끝

public class 연습 {                              // 원래 이름 BookArray
    public static void main(String[] args) {    // 프로그램 시작
        Book[] book = new Book[2];              // ① 레퍼런스 칸 2개만 생김 (값 null · 객체는 아직 없음)
        Scanner sc = new Scanner(System.in);    // 입력 준비

        for (int i = 0; i < book.length; i++) { // i = 0, 1 → 두 번
            System.out.print("제목>>");          // 안내 (print 라 줄 안 바꿈)
            String title = sc.nextLine();       // 한 줄 통째로 → "사랑의 기술" (띄어쓰기 포함)
            System.out.print("저자>>");          // 안내
            String author = sc.nextLine();      // "에리히 프롬"
            book[i] = new Book(title, author);  // ② 칸마다 Book 객체를 만들어 연결
        }                                       // for 끝

        for (int i = 0; i < book.length; i++) { // 두 권 출력
            System.out.print("(" + book[i].title + ", " + book[i].author + ")");   // ③ (제목, 저자) — print 라 한 줄에 이어서
        }                                       // for 끝
        sc.close();                             // 입력 닫기 (교재 예제처럼)
    }                                           // main 끝
}                                               // 연습 끝
