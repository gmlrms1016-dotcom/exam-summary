// E00-condition/ex01_00-01if1.c (수업 코드) — if ~ else if
#define _CRT_SECURE_NO_WARNINGS   // Visual Studio 에서 scanf 보안 경고(C4996) 끄기 — 맨 위에 써야 함
#include <stdio.h>                // printf · scanf 를 쓰기 위한 표준 입출력 헤더
int main() {   // () = 괄호를 비워 둔 옛 방식 · int main(void) 는 '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int a = 10;                   // a = 10
    int b = 20;                   // b = 20
    if (a > b) {                       // 10 > 20 → 거짓(0) → 건너뜀
        printf("a는 b보다 크다");      // 실행 안 됨 (조건이 거짓)
    }                             // if 끝
    else if (a < b) {                  // 10 < 20 → 참(1) → 실행
        printf("a는 b보다 작다");        // \n 이 없다 → 출력 뒤 줄이 안 바뀜
    }                             // if 끝
    return 0;                     // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                 // main 함수 끝
a는 b보다 작다

    printf("절대값: %d ", (a > 0) ? a : -a);       // 음수면 - 를 붙여 양수로

    printf("%s\n", ((b % 2) == 0) ? "짝수" : "홀수");   // %s = 문자열

    return 0;                                   // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                               // main 함수 끝

// 17loopsum.c — 1에서 입력된 정수까지의 합 구하기 (do ~ while 안에 중첩 for)
#define _CRT_SECURE_NO_WARNINGS           // Visual Studio 에서 scanf 보안 경고(C4996) 끄기 — 맨 위에 써야 함
#include <stdio.h>                        // printf · scanf 를 쓰기 위한 표준 입출력 헤더

int main(void) {   // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int input, sum, i, j;                 // 입력값 · 합 · 반복 변수 두 개
    do {                                  // 먼저 한 번 실행하고 끝에서 조건 검사
        printf("양의 정수 또는 0(종료)을 입력: ");   // 입력 안내 (줄바꿈 없음)
        scanf("%d", &input);              // 정수 입력 → input (예: 3)

        for (i = 1; i <= input; i++) {             // 1 줄 ~ input 줄
            for (j = 1, sum = 0; j <= i; j++) {    // 1 + 2 + … + i
                printf("%d", j);          // 더할 수 j (줄바꿈 없음)
                j == i ? printf(" = ") : printf(" + ");   // 마지막 수 뒤에는 =
                sum += j;                 // sum = sum + j
            }                             // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
            printf("%d\n", sum);          // 그 줄의 합 → 1 = 1 / 1 + 2 = 3 / 1 + 2 + 3 = 6
        }                                 // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
    } while (input > 0);               // 0 이하면 종료

    puts("종료합니다.");                       // 0 이하를 넣어 끝나면 → 종료합니다.

    return 0;                             // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                         // main 함수 끝
