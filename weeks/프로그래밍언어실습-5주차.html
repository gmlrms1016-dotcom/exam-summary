// E00-condition/ex01_00-01if1.c (수업 코드) — if ~ else if
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
int main() {
    int a = 10;
    int b = 20;
    if (a > b) {                       // 10 > 20 → 거짓(0) → 건너뜀
        printf("a는 b보다 크다");
    }
    else if (a < b) {                  // 10 < 20 → 참(1) → 실행
        printf("a는 b보다 작다");        // \n 이 없다 → 출력 뒤 줄이 안 바뀜
    }
    return 0;
}

// Quiz 8 (수업 코드) — 키보드로 세 수를 입력받아 가장 큰 값 출력 (중첩 if)
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
int main() {
    int x, y, z;
    printf("세 수를 입력하세요? ");
    scanf("%d%d%d", &x, &y, &z);       // 스페이스 · 엔터로 구분해서 3개
    if (x > y) {                       // x 가 더 크면 → 이제 x 와 z 만 비교
        if (x > z) {
            printf("max : %d", x);
        }
        else
            printf("max : %d", z);
    }
    else {                             // y 가 x 이상이면 → y 와 z 만 비교
        if (y > z) {
            printf("max : %d", y);
        }
        else {
            printf("max : %d", z);
        }
    }
    return 0;
}

3. switch ~ case · break ★시험

🔥 필기 [시험] — switch case 에 break 를 빼면? → 그 case 에서 안 멈추고 밑의 case 까지 계속 실행돼서 아래 printf 들도 출력된다 · break = switch 블록을 빠져나오는 문장 · 어느 case 에도 맞지 않으면 default
부분뜻
switch (식)식의 값(정수 · 문자)을 보고
case 값:그 값과 같은 case 로 이동해서 거기부터 실행
break;switch 블록을 빠져나감 — 없으면 아래 case 로 계속
default:맞는 case 가 없을 때 (생략 가능 · 위치 자유)

예제 ① ★ break 가 있을 때 vs 없을 때 (필기 시험 내용 확인)

// ★시험 — switch 에서 break 를 빼면? (필기: 안 멈추고 밑의 case 까지 실행)
#include <stdio.h>
int main() {
    int n = 2;

    printf("① break 있음\n");
    switch (n) {
    case 1:
        printf("one\n");
        break;
    case 2:                            // n == 2 → 여기로 이동
        printf("two\n");

#include <stdio.h>
int main() {
    int i;
    for (i = 1; i <= 3; i++)
        printf("%d ", i);
    printf("end");
    i = 1;
    while (i <= 2) {
        printf("i == %d\n", i);
        i++;
    }
    printf("종료..");
    return 0;
}
