## 2. 두더지잡기 게임 — 1초마다 랜덤 위치 · 누르면 점수 (1110_01 source.html)
> 📝 **교수님 메모** 원본 파일 맨 위 `<!--원본 … -->` 주석 안 = HTML · CSS 뼈대 + **빈 `<script>`** → 수업에서 script 를 채우고 **클릭위치** 줄(`#xy_board`)을 더했어요.

```html
<!DOCTYPE html>                                                  <!-- HTML5 문서 -->
<html lang="ko">                                                 <!-- 문서 시작 · 한국어 -->
<head>                                                           <!-- 문서 정보 -->
  <meta charset="UTF-8">                                         <!-- 한글이 안 깨지게 -->
  <title>두더지잡기 게임</title>                                    <!-- 탭 제목 -->
  <style>                                                        /* CSS 시작 */
    body {                                                       /* 페이지 전체 */
      font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;   /* 글꼴 (앞에서부터 있는 것) */
      background-color: #f0f2f5;                                 /* 연한 회색 배경 */
      margin: 0;                                                 /* 바깥 여백 없앰 */
      padding: 20px;                                             /* 안쪽 여백 20px */
      display: flex;                                             /* 자식들을 flex 로 배치 */
      flex-direction: column;                                    /* 위 → 아래로 */
      align-items: center;                                       /* 가로 가운데 */
    }                                                            /* body 끝 */
    #score-board {                                               /* 점수 줄 */
      font-size: 24px;                                           /* 큰 글자 */
      font-weight: bold;                                         /* 굵게 */
      margin-bottom: 20px;                                       /* 아래 여백 */
      color: #0056b3;                                            /* 파란 글자 */
    }                                                            /* #score-board 끝 */
    #game-board {                                                /* 게임 판 */
      width: 600px;                                              /* 폭 600px */
      height: 400px;                                             /* 높이 400px */
      background-color: #222;                                    /* 검은 판 */
      border-radius: 10px;                                       /* 모서리 둥글게 */
      box-shadow: 0 4px 10px rgba(0, 0, 0, 0.2);                 /* 그림자 */
      position: relative;                                        /* ★ 안의 absolute 요소가 이 판을 기준으로 위치 */
      overflow: hidden;                                          /* 판 밖으로 나간 부분은 안 보이게 */
    }                                                            /* #game-board 끝 */
    #target {                                                    /* 두더지(빨간 네모) */
      width: 50px;                                               /* 폭 50px */
      height: 50px;                                              /* 높이 50px */
      background-color: #ff4757;                                 /* 빨강 */
      border-radius: 8px;                                        /* 모서리 둥글게 */
      cursor: pointer;                                           /* 손가락 커서 */
      box-shadow: 0 0 10px rgba(255, 71, 87, 0.8);               /* 빨간 빛 */
      position: absolute;                                        /* ★ 판(relative) 기준으로 left · top 자리에 */
      top: 0;                                                    /* 처음엔 판 맨 위 */
      left: 0;                                                   /* 처음엔 판 맨 왼쪽 */
      transition: transform 0.1s;                                /* transform 이 바뀔 때 0.1초 동안 부드럽게 (이 코드는 transform 을 안 바꿈) */
    }                                                            /* #target 끝 */
  </style>                                                       <!-- CSS 끝 -->
</head>                                                          <!-- 문서 정보 끝 -->
<body>                                                           <!-- 화면에 보이는 부분 -->
  <h1>두더지잡기 게임</h1>                                         <!-- 제목 -->
  <div id="score-board">점수: <span id="score">0</span></div>    <!-- 점수 — 숫자 칸만 span 으로 -->
  <div id="xy_board">클릭위치 : <span id="xy"></span></div>       <!-- 오늘 더한 줄 — 판을 누른 위치 -->
  <div id="game-board">                                          <!-- 게임 판 -->
    <div id="target"></div>                                      <!-- 두더지 — 판 안에 -->
  </div>                                                         <!-- 판 끝 -->
<script>                                                         // 원본은 여기가 비어 있었음 → 수업에서 채움
  var gameBoard = document.getElementById('game-board');         // 판 요소
  var target = document.getElementById('target');                // 두더지 요소
  var scoreDisplay = document.getElementById('score');           // 점수 숫자 칸
  var xyDisplay = document.getElementById('xy');                 // 클릭 위치 칸
  var score=0;                                                   // 점수 0 에서 시작

  function moveTarget(){                                         // 두더지를 랜덤 위치로 옮기는 함수
    var maxX=gameBoard.clientWidth-target.clientWidth;           // 600 - 50 → 550 (넘치지 않는 가장 큰 left)
    var maxY=gameBoard.clientHeight-target.clientHeight;         // 400 - 50 → 350 (가장 큰 top)
    var randomX=Math.floor(Math.random()*maxX);                  // 0 이상 1 미만 × 550 → 0 ~ 549.99… → 내림 → 0 ~ 549
    var randomY=Math.floor(Math.random()*maxY);                  // 같은 방법 → 0 ~ 349
    target.style.left=randomX+'px';                              // 숫자 + 'px' → 예: "123px" (단위가 없으면 무시됨)
    target.style.top=randomY+'px';                               // 예: "45px"
    target.style.display='block';                                // 숨었던 두더지를 다시 보이게
  };                                                             // moveTarget 끝 (뒤의 ; 는 없어도 됨)

  target.addEventListener('click',function(){                    // 두더지를 누르면
    score++;                                                     // 점수 1 올림
    scoreDisplay.innerText=score;                                // 화면 점수 칸에 새 점수
    target.style.display='none';                                 // 맞힌 두더지는 숨김 → 다음 moveTarget 때 다시 나타남
  });                                                            // 두더지 클릭 끝

  gameBoard.addEventListener('click',function(e){                // 판을 누르면 (e = 이벤트 객체)
    xyDisplay.innerHTML="x:"+e.clientX+", y:"+e.clientY;         // 누른 곳의 좌표 → 예: "x:123, y:45"
  });                                                            // 판 클릭 끝
  setInterval(moveTarget,1000);                                  // 1000ms(1초)마다 moveTarget 을 되풀이 — 함수 이름만 (괄호 X)
</script>                                                        <!-- 스크립트 끝 -->
</body>                                                          <!-- 화면 끝 -->
</html>                                                          <!-- 문서 끝 -->
```

> ▶ 크롬에서 직접 확인 — 판 600 × 400 · 두더지 50 × 50 → `moveTarget()` 을 300번 불러 보니 left 는 **0 ~ 549px**, top 은 **0 ~ 349px** 안에서만 바뀌어요 (판 밖으로 안 나감) · 두더지를 누르면 점수 **1** · `display` 가 `none` → 다음 1초 뒤 `block` 으로 다시 나타나요 · 판을 누르면 `클릭위치 : x:123, y:45` 처럼 좌표가 찍혀요.

| 핵심 | 뜻 · 확인 |
|---|---|
| `position: relative` (판) + `absolute` (두더지) | 두더지의 `left` · `top` 이 **판 왼쪽 위**를 기준으로 계산돼요 · `overflow: hidden` = 판 밖은 잘라서 안 보이게 |
| `clientWidth` · `clientHeight` | 요소 안쪽 폭 · 높이 (px 숫자) → `판 - 두더지` 를 최댓값으로 써야 두더지가 판 밖으로 안 나가요 |
| `Math.random()` | **0 이상 1 미만**의 실수 (1 은 안 나옴) |
| `Math.floor(x)` | 소수점 아래를 **버림**(내림) → `Math.floor(Math.random() * 550)` = 0 ~ 549 정수 |
| `style.left = 값 + 'px'` | 🤖 크롬 확인 — `style.left = 100` 처럼 **단위 없이** 숫자만 넣으면 **무시**돼요(값이 안 바뀜) → `'px'` 를 꼭 붙여요 |
| `style.display` | `'none'` = 숨김 · `'block'` = 보임 |
| `setInterval(함수, 1000)` | 1000ms(1초)마다 함수를 **되풀이** (6주차 타이머) · 🤖 `setInterval(moveTarget(), 1000)` 처럼 괄호를 붙이면 지금 **한 번만** 실행되고 되풀이되지 않아요 (크롬 확인) |
| `e.clientX` · `e.clientY` | 누른 곳의 좌표 — 🤖 **브라우저 창 왼쪽 위** 기준이라 판 안 좌표와는 달라요 |
| 🤖 이벤트 버블링 | 두더지는 판 **안**에 있어서 두더지를 누르면 판의 click 도 같이 실행 → 점수와 클릭위치가 함께 바뀌어요 (크롬 확인) |

## 3. ✍️ 할 일 목록 — 선택 삭제 + for…of 를 map 으로 (1027_01_event.html)
> 📝 **교수님 메모** 파일 맨 위 주석 “저번주차 소스코드(저번주차 원본 참고)” — 10주차 [1번 코드](웹프로그래밍-10주차.html)에 **선택 수정 · 선택 삭제** 버튼을 더하고, 누른 항목 표시를 지우는 반복을 **map** 으로 바꿨어요.

```html
<!--저번주차 소스코드(저번주차 원본 참고)-->                         <!-- 교수님 메모 그대로 -->
<!DOCTYPE html>                                                  <!-- HTML5 문서 -->
<html>                                                           <!-- 문서 시작 -->
<head>                                                           <!-- 문서 정보 -->
    <meta charset="UTF-8">                                       <!-- 한글 -->
    <meta name="viewport" content="width=device-width, initial-scale=1.0">   <!-- 폰 화면 폭에 맞춤 -->
    <title>Document</title>                                      <!-- 탭 제목 -->
    <style>                                                      /* CSS 시작 */
        #add-btn{ padding: 5px 10px; border: 0; background: #f80; color: white; border-radius: 5px; }      /* 추가 버튼 — 주황 */
        #edit-btn{ padding: 5px 10px; border: 0; background: #38c; color: white; border-radius: 5px; }     /* 수정 버튼 — 파랑 */
        #delete-btn{ padding: 5px 10px; border: 0; background: #e33; color: white; border-radius: 5px; }   /* 삭제 버튼 — 빨강 */
        ul{ padding: 0; list-style-position: inside; }           /* 목록 — 왼쪽 들여쓰기 없앰 */
        li{ border-bottom: 1px solid #999; padding: 5px 0; }     /* 항목마다 아래 회색 줄 */
        .active{ background: #abc; }                             /* 누른 항목 — 연한 파랑 */
    </style>                                                     <!-- CSS 끝 -->
</head>                                                          <!-- 문서 정보 끝 -->
<body>                                                           <!-- 화면 -->
    <h1 id='title' >할일 목록</h1>                                 <!-- 제목 — 누른 항목 글자로 바뀜 -->
    <button id="add-btn">목록 추가</button>                         <!-- 추가 -->
    <button id="edit-btn">선택 수정</button>                        <!-- 오늘 더함 — 아직 기능 없음 -->
    <button id="delete-btn">선택 삭제</button>                      <!-- 오늘 더함 -->
    <ul id='list'>                                               <!-- 목록 -->
        <li>제목 1</li>                                          <!-- 항목 1 -->
        <li>제목 2</li>                                          <!-- 항목 2 -->
        <li>제목 3</li>                                          <!-- 항목 3 -->
        <li>제목 4</li>                                          <!-- 항목 4 -->
    </ul>                                                        <!-- 목록 끝 -->
    <script>                                                     // 스크립트 시작
        let title = document.getElementById('title');            // h1
        let addBtn = document.getElementById('add-btn');         // 추가 버튼
        let editBtn = document.getElementById('edit-btn');       // 수정 버튼 (변수만 · 리스너 없음)
        let deleteBtn = document.getElementById('delete-btn');   // 삭제 버튼
        let list = document.getElementById('list');              // ul
        deleteBtn.addEventListener('click',()=>{                 // 삭제 버튼을 누르면 (화살표 함수)
            var selectedItem=document.querySelector("#list .active");   // #list 안에서 class 가 active 인 첫 요소 · 없으면 null
            if (selectedItem){                                   // 찾았으면 (null 이면 falsy → else)
                selectedItem.remove();                           // 그 li 를 문서에서 지움
            }                                                    // if 끝
            else {                                               // 아무것도 안 골랐으면
                alert('삭제할 항목을 먼저 클릭해서 선택해주세요.');      // 알림 창
            }                                                    // else 끝
        });                                                      // 삭제 끝
        list.addEventListener('click', activeItem);              // ul 하나에 리스너 (이벤트 위임 · 10주차)
        function activeItem(e){                                  // 누른 항목 표시 함수
           console.log(e);                                       // 이벤트 객체 확인용
           if(e.target.nodeName == 'LI'){                        // 실제로 누른 게 li 일 때만
                title.innerHTML = e.target.innerText;            // 제목 ← 누른 항목 글자 (10주차는 innerHTML)
                let liArray=e.target.parentNode.children;        // 부모 ul 의 자식들 = HTMLCollection (배열 아님)
                /*for문을 map으로 바꾸기
                for( let item of liArray ){
                    item.removeAttribute('class');
                }
                liArray.map((it)=>{
                    it.removeAttribute('class');
                });

                Array.form(liArray).map((it)=>{
                    it.removeAttribute('class');
                });
                */                                               // ↑ 수업에서 차례로 써 본 것 (주석) — 아래 표 참고
                [...liArray].map((it)=>{                         // spread 로 진짜 배열로 펼친 뒤 map — 하나씩
                    it.removeAttribute('class');                 // 모든 li 의 class 지움 (파란 표시 해제)
                });                                              // map 끝
                e.target.setAttribute('class', 'active');        // 방금 누른 li 만 active → 파란 표시
           }                                                     // if 끝
        };                                                       // activeItem 끝
        addBtn.addEventListener('click' ,function(){             // 추가 버튼 (10주차 그대로)
           let txt=prompt("할 일  입력");                          // 입력 창
           list.innerHTML += '<li>'+txt+'</li>';                 // 목록 끝에 li 추가
        } );                                                     // 추가 끝
    </script>                                                    <!-- 스크립트 끝 -->
</body>                                                          <!-- 화면 끝 -->
</html>                                                          <!-- 문서 끝 -->
```

> ▶ 크롬에서 직접 눌러 본 결과 — 아무것도 안 고르고 **선택 삭제** → 알림 `삭제할 항목을 먼저 클릭해서 선택해주세요.` · **제목 3** 을 누르면 h1 이 `제목 3` · 제목 3 만 파란 표시 · 이어서 **제목 1** 을 누르고 **선택 삭제** → 목록이 `제목 2 · 제목 3 · 제목 4` (h1 은 `제목 1` 그대로) · **선택 수정** 은 아직 리스너가 없어서 눌러도 아무 일도 없어요.

| 주석 속 시도 (수업 순서) | 크롬 결과 | 왜 |
|---|---|---|
| `for (let item of liArray)` | 됨 (10주차 코드) | HTMLCollection 도 for…of 로 돌 수 있어요 |
| `liArray.map(…)` | `TypeError: liArray.map is not a function` | `children` = **HTMLCollection** — 배열처럼 생겼지만(유사 배열) **배열 메서드(map)가 없어요** |
| `Array.form(liArray)` | `TypeError: Array.form is not a function` | ⚠️ **오타** — `Array.from` 이 맞아요 (`Array.from(liArray)` → 진짜 배열 → map 됨) |
| `[...liArray].map(…)` | 됨 (오늘 최종 코드) | `...` **spread**(7주차)로 펼쳐 **새 배열**을 만든 뒤 map |

> 🤖 여기서 map 은 돌려받은 새 배열을 안 쓰고 “하나씩 실행”만 해서, 같은 일은 **forEach** 가 더 알맞아요 (관례 · 결과는 같음) — map 은 아래 4번처럼 **새 배열을 만들 때** 써요.

## 4. ✍️ 앞으로 많이 쓸 함수 — 배열 변형 메서드 filter · map (1110_02.js)
✍️ **필기** (수업 파일 주석 그대로) — “앞으로 많이 쓸 함수 5개 · 배열 변형 메서드 · 1. **filter** : 기존 배열에서 조건을 만족하는 요소들만 필터링하여 **새로운 배열로 반환** · 2. **map** : 배열의 모든 요소를 순회하면서, 각각 콜백함수를 실행하고 그 결과값들을 모아서 **새로운 배열로 반환**”

- 🤖 “5개” 중 오늘은 filter · map **2개**만 적었어요 — 나머지 3개는 다음 수업에서 (확인 못 함)
- ⚠️ 수업 파일을 그대로 실행하면 2번째 줄 `arr2.map(…)` 에서 `ReferenceError: Cannot access 'arr2' before initialization` — `let arr2` 가 **아래 줄**에 있어서 선언 전에 쓴 것 (let 은 선언 전에 못 씀) · `arr1.map` 의 오타로 보여요 (node 로 확인)

```js
// 1110_02.js 를 고쳐서 바로 실행되게 — map 과 filter 는 원본을 안 바꾸고 새 배열을 돌려줌
let arr1 = [1, 2, 3];                                    // 원본 배열
let newMapArr1 = arr1.map((item) => {                    // map: 요소를 하나씩 콜백에 넣음 (item = 1, 2, 3)
    return item * 2;                                     // 1→2 · 2→4 · 3→6 → 결과를 모아 새 배열
});                                                      // map 끝
console.log(newMapArr1);                                 // [ 2, 4, 6 ]
console.log(arr1);                                       // [ 1, 2, 3 ] — 원본은 그대로
let nums = [1, 2, 3, 4, 5];                              // filter 연습용 배열
let odd = nums.filter((item) => {                        // filter: 콜백이 true 를 돌려준 요소만 남김
    return item % 2 === 1;                               // 1 % 2 → 1 → true · 2 % 2 → 0 → false …
});                                                      // filter 끝
console.log(odd);                                        // [ 1, 3, 5 ]
let big = nums.filter((item) => item > 3);               // 한 줄 화살표 함수 — { return } 생략: 4 > 3 · 5 > 3 만 true
console.log(big);                                        // [ 4, 5 ]
console.log(nums.map((item) => item > 3));               // map 은 콜백 결과를 그대로 모음 → true/false 배열
```
```output
[ 2, 4, 6 ]
[ 1, 2, 3 ]
[ 1, 3, 5 ]
[ 4, 5 ]
[ false, false, false, true, true ]
```

| 메서드 | 콜백이 돌려준 값으로 | 결과 | 원본 |
|---|---|---|---|
| **map** | 그 값을 **그대로 모음** | 길이가 **같은** 새 배열 | 그대로 |
| **filter** | **true 인 요소만** 남김 | 길이가 **같거나 짧은** 새 배열 | 그대로 |
| forEach (10주차) | 쓰지 않음 | `undefined` (하나씩 실행만) | 그대로 |
