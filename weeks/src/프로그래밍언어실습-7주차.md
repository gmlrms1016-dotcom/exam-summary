### ① 구구단(for문) — 입력: 2
```c
// 구구단(for문)                                   // 단을 입력받아 1 ~ 9 를 곱해 출력
#define _CRT_SECURE_NO_WARNINGS                    // VS 에서 scanf 경고 끄기
#include <stdio.h>                                 // printf · scanf

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int step, i;                                   // step = 단 · i = 곱할 수 (1 ~ 9)
    printf("구구단의 단을 입력하세요: ");          // 안내 (줄바꿈 없음 → 입력이 같은 줄에)
    scanf("%d", &step);                            // 입력 2 → step = 2

    for (i = 1; i <= 9; i++) {                     // i = 1 → 2 → … → 9 (10 이 되면 끝)
        printf("%d * %d = %d\n", step, i, step * i);   // 2 * 1 = 2 → 2 * 2 = 4 → … → 2 * 9 = 18
    }                                              // for 끝 → i++ 하고 조건 검사
    return 0;                                      // 정상 종료
}                                                  // main 끝
```

### ② 구구단(while) — 입력: 2
```c
// 구구단(while)                                   // for 를 while 로 — 초기값 · 조건 · 증가를 따로 씀
#define _CRT_SECURE_NO_WARNINGS                    // scanf 경고 끄기
#include <stdio.h>                                 // printf · scanf

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
     int step, i;                                  // step = 단 · i = 곱할 수
     printf("구구단의 단을 입력하세요: ");         // 안내
     scanf("%d", &step);                           // 입력 2 → step = 2

     i = 1;                                        // ① 초기값

}                                                  // main 끝
```

### ③ 구구단(do~while) — 입력: 2
```c
// 구구단(do~while)                                // 먼저 한 번 실행하고 나서 조건 검사
#define _CRT_SECURE_NO_WARNINGS                    // scanf 경고 끄기
#include <stdio.h>                                 // printf · scanf

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int step, i;                                   // step = 단 · i = 곱할 수
    printf("구구단의 단을 입력하세요: ");          // 안내
    scanf("%d", &step);                            // 입력 2 → step = 2

    i = 1;                                         // 초기값
    do {                                           // 조건을 보기 전에 무조건 한 번 실행
        printf("%d * %d = %d\n", step, i, step * i);   // 2 * i 출력
        i++;                                       // 증가
    } while (i <= 9);                              // 끝에서 검사 — 세미콜론(;) 꼭
    return 0;                                      // 정상 종료
}                                                  // main 끝
```
```output

2 * 4 = 8
2 * 5 = 10
2 * 6 = 12
```

| 반복문 | 조건 검사 | 최소 실행 횟수 | 끝에 `;` |
|---|---|---|---|
| `for (초기; 조건; 증가)` | 먼저 | 0번 | 없음 |
| `while (조건)` | 먼저 | 0번 | 없음 |
| `do { } while (조건);` | **나중** | **1번** | **있음** ★ |

### ④ 🔥 구구단(break 활용) — 입력: 12 → 3
```c
// 구구단(break 활용)                              // 1 ~ 9 가 들어올 때까지 다시 묻기
#define _CRT_SECURE_NO_WARNINGS                    // scanf 경고 끄기
#include <stdio.h>                                 // printf · scanf

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int step, i;                                   // step = 단 · i = 곱할 수

    while (1) {                                    // 1 = 항상 참 → 무한 반복
        printf("1부터 9사이의 숫자를 입력하세요: ");   // 안내
        scanf("%d", &step);                        // 첫 입력 12 → 범위 밖 / 둘째 입력 3 → 범위 안

        if (step >= 1 && step <= 9) {              // 1 이상 그리고(&&) 9 이하면
            break;                                 // while 을 빠져나감 (12 는 통과 못 함 → 다시 물음)
        }                                          // if 끝
    }                                              // while 끝

    for (i = 1; i <= 9; i++) {                     // 3단 출력
        printf("%d * %d = %d\n", step, i, step * i);   // 3 * 1 = 3 … 3 * 9 = 27
    }                                              // for 끝

    return 0;                                      // 정상 종료
}                                                  // main 끝
```
```output

3 * 3 = 9
```

> `while (1)` = 항상 참 → **break 를 만나야만** 끝나요. 12 는 `1 ~ 9` 가 아니라서 다시 묻고, 3 이 들어오면 break → 아래 for 로.

### ⑤ 🔥 구구단(continue 활용) — 입력: 3
```c
// 구구단(continue 활용)                           // i 가 5 일 때만 건너뛰기
#define _CRT_SECURE_NO_WARNINGS                    // scanf 경고 끄기
#include <stdio.h>                                 // printf · scanf

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int step, i, j;                                // j 는 선언만 하고 안 씀
    printf("숫자를 입력하세요: ");                 // 안내
    scanf("%d", &step);                            // 입력 3 → step = 3

    for (i = 1; i <= 9; i++) {                     // i = 1 ~ 9
        if (i == 5) {                              // i 가 5 이면
            continue;                              // 아래 printf 를 건너뛰고 i++ 로 → 3 * 5 줄이 없음
        }                                          // if 끝
        printf("%d * %d = %d\n", step, i, step * i);   // 3 * i 출력
    }                                              // for 끝
    return 0;                                      // 정상 종료
}                                                  // main 끝
```
```output
숫자를 입력하세요: 3
3 * 1 = 3
3 * 2 = 6
3 * 3 = 9
3 * 4 = 12
3 * 6 = 18
3 * 7 = 21
3 * 8 = 24
3 * 9 = 27
```
> `continue` = **이번 바퀴만** 건너뛰고 `i++` 로 → `3 * 5 = 15` 줄만 빠져요 (반복은 계속). `break` 였다면 `3 * 4 = 12` 까지만 찍히고 끝나요.

### 🤖 switch 안의 break vs for 안의 break — ✍️ “완전 다르다”
```c
// switch 안의 break vs for 안의 break             // ✍️ "switch 와 for 안에 break 는 완전 다르다"
#include <stdio.h>                                 // printf

int main(void) {   // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int i;                                         // 반복 변수

    for (i = 1; i <= 5; i++) {                     // i = 1 → 2 → 3 → 4 (4 에서 끝남)
        switch (i) {                               // i 로 case 고르기
        case 2:                                    // i 가 2 이면
            printf("[switch] i=2 에서 break\n");  // 출력하고
            break;                                 // switch 만 빠져나감 → for 는 계속 (다음 줄의 if 로)
        default:                                   // 2 가 아니면
            printf("i=%d\n", i);                  // i=1 · i=3 · i=4
        }                                          // switch 끝
        if (i == 4) break;                         // for 안의 break → 반복 전체가 끝 (i++ 도 안 함)
    }                                              // for 끝

    printf("반복 끝: i = %d\n", i);               // 5 까지 못 가고 i = 4
    return 0;                                      // 정상 종료
}                                                  // main 끝
```
```output
i=1
[switch] i=2 에서 break
i=3
i=4
반복 끝: i = 4
```
| 어디의 break | 빠져나가는 곳 | 그다음 |
|---|---|---|
| **switch 안** | switch **만** | switch 다음 문장 (for 는 계속 돎) |
| **for · while 안** | 반복문 **전체** | 반복문 다음 문장 (`i++` 도 안 함) |
| switch 가 for 안에 있을 때 switch 속 break | **switch 만** | for 는 안 끝남 ★ |

#### 🤖 반대로 ① 역삼각형 (위에서부터 별이 줄어듦) — 입력: 4
```c
// 역삼각형 (강의자료에는 없음 — ✍️ "별 출력이 반대로 나오는 것도 시험에 나올 가능성")
#define _CRT_SECURE_NO_WARNINGS                    // scanf 경고 끄기
#include <stdio.h>                                 // printf · scanf
```

#### 🤖 반대로 ② 오른쪽 정렬 삼각형 (공백을 먼저) — 입력: 4
```c
// 오른쪽 정렬 삼각형 (강의자료에는 없음 — 좌우가 반대인 모양)
#define _CRT_SECURE_NO_WARNINGS                    // scanf 경고 끄기
#include <stdio.h>                                 // printf · scanf

int main(void) {   // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int step, i, j;                                // i = 줄 · j = 칸
    printf("숫자를 입력하세요: ");                 // 안내
    scanf("%d", &step);                            // 입력 4 → step = 4

    for (i = 1; i <= step; i++) {                  // 줄 1 ~ 4
        for (j = 1; j <= step - i; j++) {          // 공백 step - i 개 → 3 · 2 · 1 · 0
            printf(" ");                           // 공백 하나
        }                                          // 공백 for 끝
        for (j = 1; j <= i; j++) {                 // 별 i 개 → 1 · 2 · 3 · 4
```

### ⑫ 🔥 반복문 (7쪽) — 출력결과 맞히기
✍️ **필기** — “출력결과 맞히기 · sum 0 부터 시작 num 1 부터 시작 · 무한루프 { num 과 sum 값 각각 출력 → `sum+=num;` → sum 값 출력 → `num++;` → num 이 5 보다 크다면 `break;` } · 반복문이 끝나면 num 과 sum 값 각각 출력”
```c
// 반복문                                          // ✍️ 7쪽 반복문 문제 출력결과 맞히기
#include <stdio.h>                                 // printf

void main() {   // ⚠️ 원본 그대로 void main() — VS 는 실행되지만 표준은 int main(void) + return 0 (정상 종료를 알림)
    int sum = 0, num = 1;                          // sum 은 0 부터 · num 은 1 부터

    for (; ; ) {                     // 무한루프    // 초기 · 조건 · 증가를 비움 → 끝없이 반복 (break 로만 끝남)
        printf("num(%d) + sum(%d) = ", num, sum);  // 지금 num 과 sum 출력 (줄바꿈 없음)
        sum = sum + num;                           // sum += num → 0+1=1 → 1+2=3 → 3+3=6 → 6+4=10 → 10+5=15
        printf("%d\n", sum);                       // 더한 sum 출력하고 줄바꿈
        num++;                                     // num 1 증가 → 2 · 3 · 4 · 5 · 6
        if (num > 5) break;                        // num 이 6 이 되면 반복 끝
    }                                              // for 끝

    printf("\nResult : num = %d  sum = %d\n", num, sum);   // \n 이 앞에 있어 빈 줄 하나 → num = 6 · sum = 15
}                                                  // main 끝
```

```output
Result : num = 6  sum = 15
```
| 바퀴 | 출력 앞부분 (num · sum) | sum = sum + num | num++ | num > 5 ? |
|---|---|---|---|---|
| 1 | `num(1) + sum(0) =` | 0 + 1 = **1** | 2 | 아니오 |
| 2 | `num(2) + sum(1) =` | 1 + 2 = **3** | 3 | 아니오 |

> 함정 두 개 — ① 마지막 줄 앞 `\n` 때문에 **빈 줄**이 하나 생겨요 ② break 가 `num++` **뒤**에 있어서 끝난 뒤 `num = 6` (5 가 아님).

|---|---|---|---|

| `%d` ← `010` | 8진수 10 (앞에 0) | 010 → **001 000** → 8 | **8** |

|---|---|---|

| 10진수 → 16진수 | 자리값의 합으로 쪼개 4비트 → 10 ~ 15 = a ~ f | 10 = 8 + 2 → 1010 → **a** |

### ⑭ 🔥 증가 연산자 (7쪽)
```c
// 증가 연산자                                     // 후위 i++ = 쓰고 나서 증가 · 전위 ++i = 증가하고 나서 씀
#include <stdio.h>                                 // printf

int main() {   // 원본 그대로 int main() — 괄호를 비워 둔 옛 방식 · int main(void) = '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int i, result;                                 // i · result
    i = 10;                                        // i = 10
    result = i++;                                  // 후위: 지금 i(10)를 result 에 먼저 → result = 10 → 그다음 i = 11
    printf("i: %d, result: %d\n", i, result);      // i: 11, result: 10

    i = 10;                                        // 다시 i = 10
    result = ++i;                                  // 전위: i 를 먼저 11 로 → result = 11
    printf("i: %d, result: %d\n", i, result);      // i: 11, result: 11

    return 0;                                      // 정상 종료
}                                                  // main 끝
```

```output
i: 11, result: 11
```
> **후위** `i++` = 지금 값을 **먼저 쓰고** 나중에 1 증가 → result 10 · **전위** `++i` = **먼저 1 증가**하고 씀 → result 11 · 두 경우 모두 i 는 11.

### ⑤ 🔥 04incdecop.c — ✍️ “04incdecop 문제 나옴” (24쪽)
```c
// 04incdecop.c                                    // 전위 · 후위 증가 감소
#include <stdio.h>                                 // printf

int main(void) {   // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음

	printf("%d %d\n", m++, ++n);                   // m++ = 1 을 쓰고 m = 2 · ++n = n 을 6 으로 먼저 → 1 6
	printf("%d %d\n", m, n);                       // 2 6

	printf("%d %d\n", m, n);                       // 1 5

	return 0;                                      // 정상 종료
}                                                  // main 끝
```

```output
2 6
2 5
1 5
```

|---|---|---|---|
| 1 | `m++` (쓰고 증가) · `++n` (증가하고 씀) | **1 6** | 2 · 6 |
| 2 | `m` · `n` | **2 6** | 2 · 6 |
| 3 | `m--` (쓰고 감소) · `--n` (감소하고 씀) | **2 5** | 1 · 5 |
| 4 | `m` · `n` | **1 5** | 1 · 5 |

### 13oppriority.c — 연산자 우선순위

```c
// 13oppriority.c                                  // 괄호가 없어도 우선순위대로 → 괄호 친 식과 같은 결과
#include <stdio.h>                                 // printf

int main(void) {   // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
	int speed = 90;                                // speed = 90
	int x = 1, y = 2, z = 3;                       // x = 1 · y = 2 · z = 3
	printf("%d ", 60 <= speed && speed <= 80 + 20);        // 산술(80+20) → 관계 → && 순서 → 1 && 1 → 1
	printf("%d ", ( (60 <= speed) && (speed <= (80 + 20)) ));   // 위와 같음 → 1
	printf("%d ", x % 2 == 0 ? y + z : y * z);             // 1 % 2 = 1 → 1 == 0 거짓 → y * z = 6
	printf("%d ", (x % 2 == 0) ? (y + z) : (y * z));       // 위와 같음 → 6
	printf("%d ", speed += ++x && y - 2);                  // ++x = 2(참) && (2-2=0 거짓) → 0 → speed = 90 + 0 = 90
	printf("%d\n", speed += ( (++x) && (y - 2) ));         // ++x = 3 · y - 2 = 0 → 0 → speed = 90
	return 0;                                      // 정상 종료
}                                                  // main 끝
```
```output
1 1 6 6 90 90
```
> 우선순위: 단항(`++` `!`) > 산술 > 관계 > `&&` > `||` > 조건 `? :` > 대입 `+=` > 콤마 → 괄호를 안 쳐도 친 것과 결과가 같아요 · `speed += ++x && y - 2` → `y - 2` 가 0(거짓) → `&&` 결과 0 → speed 그대로 **90**.

### 14opassociation.c — 결합성 (대입 · 조건 = 오른쪽부터)
```c
// 14opassociation.c                               // 대입 · 조건 연산자는 오른쪽부터 · * / 는 왼쪽부터
#include <stdio.h>                                 // printf

int main(void) {   // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
	int m = 5, n = 10;                             // m = 5 · n = 10
	printf("%d\n", m += n /= 3);                   // 오른쪽 먼저: n = 10 / 3 = 3 → m = 5 + 3 = 8 → 8
	printf("%d %d\n", m, n); //8, 3                // 8 3
	printf("%d ", 3 > 4 ? 3 - 4 : 3 > 4 ? 3 + 4 : 3 * 4);   // 3 > 4 거짓 → 뒤의 조건식 → 3 > 4 거짓 → 3 * 4 = 12
	printf("%d\n", 3 > 4 ? 3 - 4 : (3 > 4 ? 3 + 4 : 3 * 4));   // 괄호로 같은 뜻 → 12
	printf("%d ", 10 * 3 / 2);                     // 왼쪽부터: 30 / 2 = 15
	printf("%d\n", 10 * (3 / 2));                  // 3 / 2 = 1 (정수 몫) → 10 * 1 = 10
	return 0;                                      // 정상 종료
}                                                  // main 끝
```
```output
8
8 3
12 12
15 10
```
