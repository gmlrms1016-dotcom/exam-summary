// ArrayEx1.java — 배열 생성 · 인덱스로 값 넣기 · 꺼내기

ArrayAccess (예제 3-7) — 입력받은 수 중 제일 큰 수

// ArrayAccess.java (예제 3-7) — 양수 5개를 배열에 저장하고 제일 큰 수 출력
import java.util.Scanner;                       // Scanner(키보드 입력)를 쓰려고 불러옴

public class 연습 {   // 원래 이름 ArrayAccess → 파일 이름도 연습.java
    public static void main(String[] args) {    // 프로그램이 시작되는 main 메소드
        Scanner sc = new Scanner(System.in);    // 키보드(System.in)로 입력받는 Scanner 객체 sc 만들기
        int intArray[] = new int[5];        // 배열 생성

        int max = 0;                        // 지금까지 가장 큰 수
        System.out.println("양수 5개를 입력하세요.");    // 안내문 출력
        for (int i = 0; i < 5; i++) {           // i = 0 → 4 → 5번 입력받기
            intArray[i] = sc.nextInt();     // 입력받은 정수를 배열에 저장
            if (intArray[i] > max)          // 더 크면
                max = intArray[i];          // max 교체
        }                                       // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        System.out.print("가장 큰 수는 " + max + "입니다.");   // 반복이 끝난 뒤의 max → 예: 1 39 78 100 99 → 가장 큰 수는 100입니다.
        sc.close();                             // 다 쓴 Scanner 닫기
    }                                           // main 끝 → 프로그램 종료
}                                               // 클래스 끝

ArrayLength (예제 3-8) — length 로 평균

// ArrayLength.java (예제 3-8) — 배열의 length 만큼 입력받아 평균 구하기
import java.util.Scanner;                       // Scanner(키보드 입력)를 쓰려고 불러옴

public class 연습 {   // 원래 이름 ArrayLength → 파일 이름도 연습.java
    public static void main(String[] args) {    // 프로그램이 시작되는 main 메소드
        int intArray[] = new int[5];            // 정수 5칸 배열 생성 (기본값 0)
        int sum = 0;                            // 합계 0 으로 시작

        Scanner sc = new Scanner(System.in);    // 키보드(System.in)로 입력받는 Scanner 객체 sc 만들기
        System.out.print(intArray.length + "개의 정수를 입력하세요>>");   // intArray.length → 5 → 5 + "개의 정수를…" → 5개의 정수를 입력하세요>>
        for (int i = 0; i < intArray.length; i++)   // i = 0 → length-1(4) → 5번 반복
            intArray[i] = sc.nextInt();         // 입력받은 정수를 i 번째 칸에 저장
        for (int i = 0; i < intArray.length; i++)   // 배열을 처음부터 다시 돌면서
            sum += intArray[i];                 // sum = sum + intArray[i] → 다섯 수의 합

        System.out.print("평균은 " + (double) sum / intArray.length);   // (double) 이 sum 에 먼저 → 실수 ÷ 5 → 소수점까지 평균
        sc.close();                             // 다 쓴 Scanner 닫기
    }                                           // main 끝 → 프로그램 종료
}                                               // 클래스 끝
5개의 정수를 입력하세요>>2 3 4 5 9
평균은 4.6
📌 (double) sum / intArray.length — sum 을 먼저 double 로 바꿔서 23.0 / 5 = 4.6 · (double) (sum / intArray.length) 로 쓰면 정수 나눗셈 23 / 5 = 4 가 먼저 돼서 4.0 (타입 변환 ★시험단골 → 3주차)

2. for-each 문 (예제 3-9)

📌 for-each 문 = 배열이나 나열(enumeration)의 원소를 처음부터 끝까지 차례로 꺼내 쓰는 for 문 · for (int k : n) → 반복할 때마다 k 는 n[0], n[1], … n[4] (인덱스가 아니라 원소 값)
쓰는 법반복할 때마다 변수에 들어가는 것
for (int k : n)int 배열 n 의 원소 n[0] ~ n[4]
for (String s : names)String 배열 names 의 원소 names[0] ~ names[5]
for (Week day : Week.values())enum Week 의 값 월, 화, … 일

foreachEx (예제 3-9)

// foreachEx.java (예제 3-9) — for-each 로 int 배열 · String 배열 · enum 돌기
public class 연습 {   // 원래 이름 foreachEx → 파일 이름도 연습.java
    enum Week { 월, 화, 수, 목, 금, 토, 일 }     // 나열(enum) 타입

    public static void main(String[] args) {    // 프로그램이 시작되는 main 메소드
        int[] n = {1, 2, 3, 4, 5};              // 정수 배열 {1,2,3,4,5}
        String names[] = {"사과", "배", "바나나", "체리", "딸기", "포도"};   // 문자열 배열 6칸

        int sum = 0;                            // 합계 0 으로 시작
        for (int i : n) {                        // i = 1, 2, 3, 4, 5
            System.out.print(i + " ");          // i 출력 + 공백 → 1 2 3 4 5 
            sum += i;                           // sum = sum + i → 15
        }                                       // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        System.out.println("합은 " + sum);        // 합은 15 출력 후 줄바꿈
        for (String s : names) {                 // s = "사과", "배", …
            System.out.print(s + " ");          // 과일 이름 + 공백 → 사과 배 바나나 체리 딸기 포도 
        }                                       // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        System.out.println();                   // 줄바꿈만
        for (Week day : Week.values()) {         // day = 월, 화, … 일
            System.out.print(day + "요일 ");      // 요일 + "요일 " → 월요일 화요일 … 일요일 
        }                                       // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        System.out.println();                   // 줄바꿈만
    }                                           // main 끝 → 프로그램 종료
}                                               // 클래스 끝

ScoreAverrage (예제 3-10) — 4년 평점 평균

// ScoreAverrage.java (예제 3-10) — 2차원 배열로 4년(8학기) 평점 평균
public class 연습 {   // 원래 이름 ScoreAverrage → 파일 이름도 연습.java
    public static void main(String[] args) {    // 프로그램이 시작되는 main 메소드
        double score[][] = {{3.3, 3.4},     // 1학년 1, 2학기
                            {3.5, 3.6},     // 2학년
                            {3.7, 4.0},     // 3학년
                            {4.1, 4.2}};    // 4학년

        double sum = 0;                         // 8학기 평점 합계 0 으로 시작
        for (int year = 0; year < score.length; year++) {             // 학년(행) 4번
            for (int term = 0; term < score[year].length; term++) {   // 학기(열) 2번
                sum += score[year][term];       // sum = sum + 그 학기 평점 → 8번 더하면 29.8
            }                                   // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        }                                       // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        int n = score.length;       // score.length → 행(학년) 개수 → 4
        int m = score[0].length;    // score[0] = {3.3, 3.4} → 길이 → 열(학기) 개수 2
        System.out.println("4년 전체 평점 평균은 " + sum / (n * m));   // 29.8 ÷ (4 × 2) → 29.8 / 8 → 4년 전체 평점 평균은 3.725
    }                                           // main 끝 → 프로그램 종료
}                                               // 클래스 끝
4년 전체 평점 평균은 3.725

IrregularArray (예제 3-11) — 비정방형 배열

// IrregularArray.java (예제 3-11) — 행마다 열 개수가 다른 2차원 배열
public class 연습 {   // 원래 이름 IrregularArray → 파일 이름도 연습.java
    public static void main(String[] args) {    // 프로그램이 시작되는 main 메소드
        int intArray[][] = new int[4][];    // 행 4개만 먼저 만들고
        intArray[0] = new int[3];           // 0행: 열 3개
        intArray[1] = new int[2];           // 1행: 열 2개
        intArray[2] = new int[3];               // 2행: 열 3개
        intArray[3] = new int[2];               // 3행: 열 2개

        for (int i = 0; i < intArray.length; i++) {   // i = 0 → 3 → 행 4개
            for (int j = 0; j < intArray[i].length; j++) {   // 행마다 길이가 달라서 intArray[i].length
                intArray[i][j] = (i + 1) * 10 + j;   // (i+1)*10 + j → 0행 10 11 12 · 1행 20 21 · 2행 30 31 32 · 3행 40 41
            }                                   // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        }                                       // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        for (int i = 0; i < intArray.length; i++) {   // 출력도 행마다
            for (int j = 0; j < intArray[i].length; j++) {   // 그 행의 열 개수만큼
                System.out.print(intArray[i][j] + " ");   // 값 + 공백 (줄바꿈 없음)
            }                                   // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
            System.out.println();               // 한 행이 끝나면 줄바꿈 → 10 11 12 / 20 21 / 30 31 32 / 40 41
        }                                       // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
    }                                           // main 끝 → 프로그램 종료
}                                               // 클래스 끝

4. 예외 처리 실습 — 0으로 나누기 (예제 3-15) ★시험단골

// DevideByZeroHandling.java (예제 3-15) — 0 으로 나누면 다시 입력받기
import java.util.Scanner;                       // Scanner(키보드 입력)를 쓰려고 불러옴

public class 연습 {   // 원래 이름 DevideByZeroHandling → 파일 이름도 연습.java
    public static void main(String[] args) {    // 프로그램이 시작되는 main 메소드
        Scanner sc = new Scanner(System.in);    // 키보드(System.in)로 입력받는 Scanner 객체 sc 만들기

        while (true) {                          // 제대로 나눌 때까지 무한 반복 (break 로만 탈출)
            System.out.print("나뉨수를 입력하시오:");    // 나뉨수 질문 (줄바꿈 없음)
            int dividend = sc.nextInt();        // 나뉨수 입력 → dividend (예: 100)
            System.out.print("나눗수를 입력하시오:");    // 나눗수 질문
            int divisor = sc.nextInt();         // 나눗수 입력 → divisor (예: 0 → 예외 / 5 → 정상)
            try {                               // 예외가 날 수 있는 나눗셈을 try 로 감쌈
                System.out.println(dividend + "를 " + divisor + "로 나누면 몫은 " + dividend / divisor + "입니다.");   // divisor 가 0 이면 여기서 ArithmeticException → catch 로 · 아니면 100를 5로 나누면 몫은 20입니다.
                break;                              // 정상으로 나누면 while 탈출
            } catch (Exception e) {                 // 0 으로 나누면 여기로 (break 는 건너뜀)
                System.out.println("에러났습니다.");  // 예외가 났을 때 안내
                System.out.println("0으로 나눌 수 없습니다! 다시 입력하세요");   // 다시 입력하라고 안내 → while 처음으로 돌아감
                e.printStackTrace();                // 예외 종류와 발생 위치(줄 번호) 출력
            }                                   // catch 끝
        }                                       // while 끝 → 조건 다시 검사 (거짓이면 빠져나옴)
        sc.close();                             // 다 쓴 Scanner 닫기
    }                                           // main 끝 → 프로그램 종료
}                                               // 클래스 끝

        cheese.price = 3000;                    // cheese 의 가격 3000
        String result3 = cheese.showInfo();     // → "치즈 붕어빵 가격은 3000원"
        System.out.println("객체 = " + result3);  // 객체 = 치즈 붕어빵 가격은 3000원
    }                                           // main 끝 → 프로그램 종료
}                                               // 클래스 끝
객체 = 팥 붕어빵 가격은 1000원
객체 = 슈크림 붕어빵 가격은 2000원
객체 = 치즈 붕어빵 가격은 3000원
📌 4장 p10 ~ 11 의 붕어빵틀 = 클래스, 구워낸 붕어빵 = 객체 를 코드로 만든 것 · 세 객체 모두 flavor · price 필드를 가지지만 값은 각자 다르다 · showInfo() 는 출력하지 않고 문자열을 return 해서, main 에서 받아 출력해요.
