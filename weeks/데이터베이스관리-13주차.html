강의자료 8쪽 [Quiz 13-01] · 실습 엑셀 Quiz1-1-join-stu “카디날리티는 무엇인가?” — 다음 두 릴레이션 R 과 S 에 대해 R 을 왼쪽, S 를 오른쪽 릴레이션으로 하고 속성 B 를 조인 속성으로 하여 자연조인, 왼쪽외부조인, 완전외부조인으로 수행하였을 때 결과 릴레이션의 카디널리티는 각각 얼마인가? · ✍️ 내 엑셀 답 3 · 4 · 5 ✅

-- 1) 자연조인 — 같은 이름의 열 B 가 같은 짝만 · B 는 한 번만
SELECT *
FROM R NATURAL JOIN S;                                       -- R.B 2 · 3 · 3 이 S 와 짝 (a · b · c) → 3행 (MySQL 은 B 를 맨 앞에: B · A · C)
-- (같은 짝을 ON 으로 쓰면 동등조인 — B 열이 2개)
SELECT *
FROM R INNER JOIN S
ON R.B = S.B;                                                -- 3행 · A · B · B · C
-- 2) 왼쪽 외부조인 — 왼쪽(R) 행은 짝이 없어도 모두
SELECT *
FROM R LEFT OUTER JOIN S
ON R.B = S.B;                                                -- 짝 3행 + d · 5 (S 에 B=5 없음) → NULL · NULL → 4행
-- (오른쪽 외부조인 — 오른쪽(S) 행은 모두)
SELECT *
FROM R RIGHT OUTER JOIN S
ON R.B = S.B;                                                -- 짝 3행 + 1 · x (R 에 B=1 없음) → NULL · NULL → 4행
-- 3) 완전 외부조인 — 양쪽 모두 (MySQL 은 FULL OUTER JOIN 이 없음 → 왼쪽 UNION 오른쪽)
SELECT * FROM R LEFT OUTER JOIN S ON R.B = S.B
UNION                                                        -- 겹친 짝 3행은 한 번만
SELECT * FROM R RIGHT OUTER JOIN S ON R.B = S.B;             -- 4 + 4 − 3 = 5행

조인카디널리티 (행 수)왜
1) 자연조인3B = 2 · 3 · 3 인 a · b · c 만 S 와 짝 — d(5) 와 S 의 1(x) 은 짝 없음
2) 왼쪽 외부조인4짝 3 + 짝 없는 왼쪽 d → d · 5 · NULL · NULL
(오른쪽 외부조인)4짝 3 + 짝 없는 오른쪽 1(x) → NULL · NULL · 1 · x

-- [실습 3-1] 중첩질의(sub query)
-- stu(sno, sname, dept, sage)
-- pro(pno, pname, dept, page)

-- 3-1) 교수 테이블에서 이정무 의 학과와 같은 학생 이름, 학과, 나이를 검색하라
SELECT sname, dept, sage
FROM stu
WHERE dept IN (SELECT dept
               FROM pro
               WHERE pname = '이정무');
-- 3-2) 교수 테이블에서 pno 가 p1 이거나 p2 인 학과와 같은 학생 이름, 학과, 나이를 검색하라
SELECT sname, dept, sage
FROM stu
WHERE dept IN (SELECT dept
               FROM pro
               WHERE pno IN('p1','p2'));
-- 3-3) 학생 테이블과 교수 테이블에서 모든 학생들보다 나이가 많은 교수의
-- 교번, 이름, 나이를 ALL 구문을 이용하여 검색하라
SELECT pno, pname, page
FROM pro
WHERE page>ALL (SELECT sage FROM stu);
-- 3-4) 학생 테이블과 교수 테이블에서 한 명 이상 교수보다 
-- 나이가 많은 학생이 있을 경우 학생의 학번, 이름, 나이를 SOME 구문으로 검색하라
SELECT sno, sname, sage
FROM stu
WHERE sage > SOME (SELECT page FROM pro);
-- 위랑 아래랑 같음
SELECT sno, sname, sage
FROM stu
WHERE sage > ANY (SELECT page FROM pro);
-- 3-5) 가장 나이가 많은 학생 이름, 나이를 검색하라
SELECT sname, sage
FROM stu
WHERE sage = (SELECT MAX(sage) FROM stu);
