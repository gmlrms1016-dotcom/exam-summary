① 🔥 기말고사 — 5쪽 [실습 1-01] 포인터 변수

✍️ 노션 “강의자료 5페이지 기말고사 예상문제”

// ex_01-pointer1.c — 📄 5쪽 [실습 1-01] 포인터 변수 (🔥 ✍️ “강의자료 5페이지 기말고사 예상문제”)
#include <stdio.h>                              // printf 를 쓰기 위한 표준 입출력 헤더

int main(void) {                                // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int i = 10;                                 // 일반 변수 i ← 값 10
    int* ptr;                                   // 포인터 변수 ptr — int 형 기억장소의 '주소'를 저장
    ptr = &i;                                   // &i = i 의 주소 → ptr 에 저장 → ptr 이 i 를 가리킴
    i = 50;                                     // i 를 직접 바꿈 → i = 50 (ptr 은 여전히 i 를 가리킴)
    *ptr = 100;                                 // *ptr = ptr 이 가리키는 곳 = i → i 에 100 저장 → i = 100
    printf("i = %d \n", i);                     // i = 100 → "i = 100 " (%d 뒤 공백도 그대로 찍힘)
    printf("*ptr = %d\n", *ptr);                // *ptr → 가리키는 i 의 값 → 100
    return 0;                                   // 0 을 돌려주며 정상 종료
}                                               // main 끝
i = 100 
*ptr = 100
  • 순서: i = 10 → ptr = &i → i = 50 → *ptr = 100 (ptr 이 가리키는 i 에 씀) → i 도 *ptr 도 100
  • 첫 줄 끝에 공백이 하나 있어요 ("i = %d \n" 의 %d 뒤 공백)

✍️ 노션 //시험문제 — printf("%d %d %d", *ptr, ptr, i)

// ✍️ 노션 “//시험문제” — 강의자료 5쪽 변형 (필기의 prt=&i; 는 ptr=&i; 오타)
#include <stdio.h>                              // printf 를 쓰기 위한 표준 입출력 헤더

int main(void) {                                // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int i;                                      // 일반 변수 i
    int *ptr;                                   // 포인터 변수 ptr
    i = 10;                                     // i ← 10
    ptr = &i;                                   // ptr ← i 의 주소
    printf("%d %d %d\n", *ptr, ptr, i);         // *ptr = 10 · ptr = 주소를 %d(10진수)로 → 실행마다 다름 · i = 10
    return 0;                                   // 정상 종료
}                                               // main 끝
10 1875614584 10
  • 첫째 · 셋째 = 10 (*ptr · i) · 가운데 = i 의 주소를 %d(10진수)로 찍은 값 → 실행마다 달라요
  • ⚠️ 필기의 prt=&i; 는 ptr=&i; 오타 (prt 는 선언하지 않은 이름 → 컴파일 오류)
  • ⚠️ 팩트체크 — 주소는 %p 로 찍는 게 맞아요. %d 에 포인터를 주면 clang 경고 format specifies type 'int' but the argument has type 'int *' (실행은 되지만 표준에서는 정의되지 않은 동작) → 시험 답은 “10 · (i 의 주소) · 10”

② 🔥 기말고사 — 수업 실행 ① 이중 포인터 (19쪽 [실습 3-01])

✍️ 노션 “오늘 한 거 실행결과 기말고사 예상문제” · ptr = &i; //여기가 중요함

// ex_05-pointer1.c — 📄 19쪽 [실습 3-01] 이중 포인터 (v1.0.1 · 🔥 ✍️ “오늘 한 거 실행결과 기말고사 예상문제”)
#include <stdio.h>                              // printf 를 쓰기 위한 표준 입출력 헤더

int main(void) {                                // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int i = 10;                                 // 일반 변수 i ← 10
    int* ptr;                                   // 포인터 ptr — int 의 주소를 저장
    int** pptr;                                 // 이중 포인터 pptr — int* 형 변수(ptr)의 주소를 저장
    ptr = &i;                                   // ✍️ “여기가 중요함” — ptr ← i 의 주소 → ptr 이 i 를 가리킴
    pptr = &ptr;                                // pptr ← ptr 의 주소 → pptr → ptr → i
    printf("*ptr = %d\n", *ptr);                // *ptr = ptr 이 가리키는 i → 10
    **pptr = 11;                                // *pptr = ptr → **pptr = *ptr = i → i 에 11 저장
    printf("i = %d, *ptr = %d, **ptr = %d\n",  i, *ptr, **pptr); // i = 11 · *ptr = i = 11 · **pptr = i = 11
    return 0;                                   // 정상 종료
}                                               // main 끝
*ptr = 10
i = 11, *ptr = 11, **ptr = 11
  • pptr → ptr → i — *pptr = ptr (i 의 주소) · **pptr = i
  • **pptr = 11; 은 i 를 바꿈 → i · *ptr · **pptr 모두 11
  • ⚠️ 강의자료가 고쳐졌어요 (v1.0.1) — 처음 올라온 판의 19쪽은 int* ptr, i = 10; 다음 줄에 int *ptr; 가 또 있고, printf 에 **ptr (ptr 은 int* 라 별 두 개 불가) → clang 컴파일 오류 3개 (redefinition of 'ptr' · indirection requires pointer operand ('int' invalid) × 2). 11/16 저녁 v1.0.1 에서 위 코드처럼 고쳐짐
  • ✍️ 수업 main.c 는 예전 판을 보고 쳐서 **pptr 로는 고쳤지만 int* ptr; 가 두 번 → clang redefinition of 'ptr' 오류 · 두 번째 줄을 지우면 실행돼요 (첫 출력 줄은 **ptr = 10)

③ 🔥 기말고사 — 수업 실행 ② 배열 포인터 (22쪽 [실습 3-03])

✍️ 노션 “강의자료 22페이지 int (*ptr)[3];” · 수업 main.c 는 %2d

// ex_07-pointer1.c — 📄 22쪽 [실습 3-03] 배열 포인터 (🔥 ✍️ 수업 main.c 는 %d 대신 %2d)
#include <stdio.h>                              // printf 를 쓰기 위한 표준 입출력 헤더

int main(void) {                                // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int nArray[2][3] = { {1,2,3}, {4,5,6} };    // 2행 3열 — 메모리에는 1 2 3 4 5 6 이 한 줄로 (행 우선)
    int (*ptr)[3];                              // 배열 포인터 — '열이 3개인 int 배열' 한 줄(12바이트)을 가리킴
    int i, j;                                   // i = 행 첨자 · j = 열 첨자
    ptr = nArray;                               // 2차원 배열 이름 = 0행(int[3])의 시작 주소 → ptr 이 0행을 가리킴
    for (i = 0; i < 2; i++) {                   // i = 0, 1 → 행 2개
        for (j = 0; j < 3; j++) {               // j = 0, 1, 2 → 열 3개
            printf("%2d", ptr[i][j]);           // ptr[i][j] = nArray[i][j] → 2칸 폭 오른쪽 정렬 (1 → " 1")
        }                                       // 안쪽 for 끝 → 한 행을 다 찍음
        printf("\n");                           // 한 행이 끝나면 줄바꿈
    }                                           // 바깥 for 끝 → 다음 행
    return 0;                                   // 정상 종료
}                                               // main 끝

  • 강의자료 그대로 printf("%d", ...) 면 띄어쓰기 없이 123 / 456 (✍️ 수업에서는 %2d → 2칸 오른쪽 정렬)
  • int (*ptr)[3] = 열이 3개인 int 배열을 가리키는 포인터 → ptr[i][j] 로 2차원 배열처럼 씀

🖥️ 교수님 엑셀 (사진) — ptr + 1 은 한 줄(12바이트) 뒤

// 🤖 배열 포인터는 +1 하면 '한 줄(열 3개 = 12바이트)' 건너뜀 — 교수님 엑셀(ptr = 100 → ptr + 1 = 112) 확인
#include <stdio.h>                              // printf 를 쓰기 위한 표준 입출력 헤더

int main(void) {                                // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int nArray[2][3] = { {1,2,3}, {4,5,6} };    // 2행 3열 배열
    int (*ptr)[3] = nArray;                     // 배열 포인터 ← 0행의 주소
    printf("%d\n", (int)sizeof(*ptr));          // *ptr = int[3] 한 줄 → 3 × 4 = 12
    printf("%d\n", (int)((char*)(ptr + 1) - (char*)ptr)); // ptr + 1 은 몇 바이트 뒤? → 12 (100 → 112)
    printf("%d %d %d\n", **ptr, *ptr[1], (*ptr)[1]); // **ptr = nArray[0][0] = 1 · *ptr[1] = *(ptr[1]) = nArray[1][0] = 4 · (*ptr)[1] = nArray[0][1] = 2
    printf("%d %d\n", (*(ptr + 1))[0], *(*(ptr + 1) + 2)); // (*(ptr+1))[0] = nArray[1][0] = 4 · *(*(ptr+1)+2) = nArray[1][2] = 6
    return 0;                                   // 정상 종료
}                                               // main 끝

  • ⚠️ 사진 속 엑셀의 112 칸 (*ptr)[1] 은 104 칸(값 2) 이에요 — 112 칸은 *ptr[1] (실행: (*ptr)[1] = 2 · *ptr[1] = 4) · 120 칸의 nArray[0][2] 는 nArray[1][2] 오타

④ 🔥 기말고사 — 수업 실행 ③ 함수 포인터 (✍️ main.c · 16쪽 요약은 빈칸)

// ✍️ 수업 main.c — 함수 포인터 (🔥 오늘 실행결과 · 📄 16쪽 [요약] 함수 포인터는 빈칸 → 노션 필기)
#include<stdio.h>                               // printf 를 쓰기 위한 표준 입출력 헤더 (< 앞 공백은 없어도 됨)

int add(int i) {                                // int 하나를 받아 int 를 돌려주는 함수
    return i+1;                                 // 받은 값 + 1 을 돌려줌
}                                               // add 끝

int main(void) {                                // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int(*pfunc)(int);                           // 함수 포인터 — (int 를 받고 int 를 돌려주는) 함수의 주소를 저장
    int result;                                 // 호출 결과를 담을 변수
    pfunc=add;                                  // 함수 이름 add = 함수의 시작 주소 → pfunc 이 add 를 가리킴
    result = pfunc(11);                         // pfunc(11) = add(11) 호출 → 11 + 1 → 12
    printf("result = %d\n", result);            // result = 12
    return 0;                                   // 정상 종료
}                                               // main 끝
result = 12
  • ✍️ 함수 포인터 = 함수의 기억장소 주소를 저장하고 그 주소의 함수를 호출 · 반환할 자료형과 매개변수 형식이 모두 같아야 · 함수 포인터는 이름만 대신하는 것
  • int (*pfunc)(int) 괄호가 꼭 필요 — 🤖 int *pfunc(int); 는 “int* 를 돌려주는 함수 선언”이 돼서 pfunc = add; 가 오류 (clang non-object type 'int *(int)' is not assignable)

⑤ 🔥 기말고사 — 수업 실행 ④ void 형 포인터 (24쪽 [실습 3-04])

// ex_09-pointer1.c — 📄 24쪽 [실습 3-04] void 형 포인터 (🔥 오늘 실행결과 · ✍️ 수업 main.c 와 같음)
#include <stdio.h>                              // printf 를 쓰기 위한 표준 입출력 헤더

int main(void) {                                // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    void* vptr;                                 // void 형 포인터 — 어떤 자료형의 주소든 저장 (가리키는 '형'은 모름)
    int i = 10;                                 // int 변수
    char ch = 'a';                              // char 변수
    float fr = 3.14f;                           // float 변수 (f = float 상수)
    vptr = &i;                                  // int 의 주소 저장 → 형변환 없이 대입 가능
    printf("%d\n", *((int*)vptr));              // (int*) 로 형변환 → int 로 읽음 → 10
    vptr = &ch;                                 // 이번엔 char 의 주소 저장
    printf("%c\n", *((char*)vptr));             // (char*) 로 형변환 → 'a'
    vptr = &fr;                                 // 이번엔 float 의 주소 저장
    printf("%f\n", *((float*)vptr));            // (float*) 로 형변환 → 3.14 → %f 는 소수 6자리 → 3.140000
    return 0;                                   // 정상 종료
}                                               // main 끝

  • ✍️ void 형 포인터 = 모든 자료형의 주소를 저장 · 단, 쓸 때 반드시 형변환 (type casting)
  • ⚠️ 팩트체크 — 형변환 없이 *vptr 하면 clang 오류 argument type 'void' is incomplete (가리키는 곳의 크기를 몰라서) → “반드시” 맞음
  • %f 는 소수 6자리 → 3.140000

⑥ ✍️ 포인터 연산 — ++ptr 이 1004번지 · 1001번지 (11쪽 [실습 2-01])

✍️ 노션 “printf("%p",++ptr); 출력이 어떻게 하면 1004번지가 나오는지, 어떻게 하면 1001번지가 나오는지 공부하기” · 📄 11쪽 “포인터 연산 (포인터 변수의 자료형 크기에 맞춰 증감)”

// ✍️ 노션 “++ptr 이 1004번지 · 1001번지가 나오는 방법 공부하기” · 📄 11쪽 [실습 2-01] 포인터 연산
#include <stdio.h>                              // printf 를 쓰기 위한 표준 입출력 헤더

int main(void) {                                // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int i = 10;                                 // int 변수 (4바이트)
    char ch = 'a';                              // char 변수 (1바이트)
    int* ip = &i;                               // int 포인터 ← i 의 주소 (1000번지라고 하면)
    char* cp = &ch;                             // char 포인터 ← ch 의 주소 (1000번지라고 하면)
    printf("%d\n", (int)((char*)(ip + 1) - (char*)ip)); // int* + 1 → sizeof(int) = 4 바이트 뒤 → 4 (1000 → 1004)
    printf("%d\n", (int)((char*)(cp + 1) - (char*)cp)); // char* + 1 → sizeof(char) = 1 바이트 뒤 → 1 (1000 → 1001)
    printf("%d %d\n", (int)sizeof(int*), (int)sizeof(char*)); // 포인터 변수 자체의 크기 → 64비트 → 8 8
    return 0;                                   // 정상 종료
}                                               // main 끝
4
1
8 8
  • int* 는 +1 하면 int 크기(4)만큼 → 1000 → 1004 · char* 는 1만큼 → 1000 → 1001
  • 포인터 변수 자체의 크기는 가리키는 자료형과 상관없이 같아요 (64비트 = 8)

📄 11쪽 그대로 — 주소는 실행마다 달라요 (이 맥에서 한 번 실행)

// ex_04-pointer1.c — 📄 11쪽 [실습 2-01] 포인터 변수 연산 (주소는 실행마다 다름)
#include <stdio.h>                              // printf 를 쓰기 위한 표준 입출력 헤더

int main(void) {                                // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int* ptr, i = 10;                           // ptr = int 포인터 · i = 10
    ptr = &i;                                   // ptr ← i 의 주소
    printf("i = %d, *ptr = %d\n", i, *ptr);     // i = 10 · *ptr = 10
    printf("ptr = %p\n", ptr);                  // i 의 주소 (16진수 · 실행마다 다름)
    ++ptr;                                      // ptr 을 1 증가 → int 크기(4)만큼 → 주소 + 4 (ptr 은 이제 i 를 안 가리킴)
    printf("ptr = %p\n", ptr);                  // 위 주소 + 4
    return 0;                                   // 정상 종료
}                                               // main 끝
i = 10, *ptr = 10
ptr = 0x16eff5b6c
ptr = 0x16eff5b70
  • 두 번째 주소 = 첫 번째 주소 + 4 (16진수 끝자리로 확인 · 예: …cac → …cb0 = ac + 4 = b0 · c(12) + 4 = 16 → 0 쓰고 1 올림)
  • ⚠️ 팩트체크 — 노션의 printf("%p",++ptr); printf("%d %d",i,*ptr); //10 10 은 아니에요. ++ptr 은 ptr 자체를 바꾸므로 그 뒤 *ptr 은 i 가 아닌 다음 칸 → 쓰레기값(정의되지 않은 동작). ptr 을 그대로 두려면 printf("%p", ptr + 1); (ptr + 1 은 계산만 · ptr 은 그대로 → *ptr = 10)
  • ⚠️ 팩트체크 — 노션 “포인터 변수의 연산: +,-,*,/,++,--” 중 * · / 는 안 돼요 (clang ptr * 2 · ptr / 2 → invalid operands to binary expression ('int *' and 'int')) · 📄 11쪽 “*, -, ++, --” 의 * 도 + 로 읽기 · 🤖 포인터끼리 빼기(q - ptr)는 됨 → 사이의 칸 수

  • ✍️ 포인터 선언 주의 — int* p; int * p; int *p; 셋 다 같음 · 두 개 이상은 int *p, *q; 처럼 각각 * (확인: int* p, q; 면 q 는 그냥 int → sizeof 가 8 · 4)
  • ✍️ 일반 변수 = 값이 저장 (int i; 4byte) · 포인터 변수 = 주소가 저장 (int *ptr;) · & 주소 연산자 · * 간접접근 연산자(역참조 연산자)

// ✍️ 노션 — 포인터 배열 활용 1: 길이가 다른 문자열 (static char *name[3] = {"HI","GOOD","MORNING"};)
#include <stdio.h>                              // printf 를 쓰기 위한 표준 입출력 헤더

int main(void) {                                // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    static char *name[3] = {"HI", "GOOD", "MORNING"}; // 포인터 배열 — 문자열 3개의 '시작 주소'만 저장 (글자는 따로)
    char name2[3][8] = {"HI", "GOOD", "MORNING"};      // 2차원 배열 — 가장 긴 "MORNING"(7 + \0 = 8칸)에 맞춰 3줄 × 8칸
    int i;                                      // 반복 첨자
    for (i = 0; i < 3; i++) {                   // i = 0 ~ 2
        printf("%s %s\n", name[i], name2[i]);   // 둘 다 같은 문자열이 찍힘
    }                                           // for 끝
    printf("%d\n", (int)sizeof(name2));         // 2차원 배열 = 3 × 8 = 24바이트 (HI 뒤 5칸 · GOOD 뒤 3칸이 빈 채 낭비)
    printf("%d\n", (int)sizeof(name));          // 포인터 배열 = 주소 3개 × 8바이트(64비트) = 24 (글자는 따로 3 + 5 + 8 = 16바이트)
    return 0;                                   // 정상 종료
}                                               // main 끝
