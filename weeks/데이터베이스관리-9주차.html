1. ✍️ SELECT 문법 요약 — 연산자 · LIKE · ORDER BY

구분기호 · 낱말뜻
관계 연산자> >= = <> != <= <초과 · 이상 · 같다 · 같지 않다 · 같지 않다 · 이하 · 미만 (6주차)
논리 연산자NOT · AND · OR아니다 · 이고 · 이거나
✍️ 패턴 LIKE% · _% = 0개 이상의 문자열 · _ = 1개 문자 · ✍️ LIKE = “~와 같은 · ~와 유사한”
✍️ 정렬 ORDER BYASC · DESC오름차순 asc = 기본값(안 써도 오름차순) · 내림차순 desc
널 값 검사IS NULL · IS NOT NULL널 값은 = 로 비교하지 않음 (아래 3-5)
✍️ 필기 — 드 모르간 법칙 ~(A ∧ B) · ~(A ∨ B) → 🤖 ~(A ∧ B) = ~A ∨ ~B · ~(A ∨ B) = ~A ∧ ~B (∧ = AND · ∨ = OR · ~ = NOT). 실습 2-7 “서울이거나 안양인 것이 아닌” = NOT (서울 OR 안양) = 서울이 아니고 AND 안양이 아닌 = NOT IN ('서울','안양') 이에요.

3. [실습1] customer — SELECT 기초 · NULL · LIKE · ORDER BY (강의자료 14 ~ 43쪽)

-- customer(cno, cname, city, point) = 고객(고객번호, 고객명, 거주지, 포인트)
CREATE  TABLE   customer (                           -- customer 테이블 만들기 시작
      cno     char(4)         NOT  NULL  ,           -- 고객번호 — 문자 4자리 · 필수
      cname  varchar(10)   NOT  NULL  ,              -- 고객명 — 최대 10자 · 필수
      city     varchar(20)  ,                        -- 거주지 — 최대 20자 · 비워도 됨
      point    int  ,                                -- 포인트 — 정수 · 비워도 됨(NULL)
      CONSTRAINT  pk_customer_cno  PRIMARY  KEY(cno) -- 제약조건 이름 pk_customer_cno 로 cno 를 기본키로
);                                                   -- CREATE TABLE 끝
insert into customer values('c101','홍길동','서울',500);    -- 1행 c101 · 홍길동 · 서울 · 500
insert into customer values('c102','임꺽정','인천',300);    -- 2행
insert into customer values('c103','박찬호','안양',800);    -- 3행
insert into customer values('c204','신동엽','과천',350);    -- 4행
insert into customer values('c205','정진우','고양',400);    -- 5행
insert into customer values('c307','정동우','서울', NULL);  -- 6행 — 새로 추가 · 포인트 없음(NULL)

-- ✍️ 내 답
SELECT * FROM customer;                              -- 1-1) 모든 열 → 6행
SELECT cno,cname,city,point FROM customer;           -- 1-2) 모든 열을 필드명으로 → 1-1 과 같음
SELECT cname,city FROM customer;                     -- 1-3) 고객명 · 거주지 두 열만
SELECT cname as 성명,city as 거주지 FROM customer;   -- 1-4) ⭐ AS = 화면에 보이는 열 이름만 바꿈 (별칭)
SELECT city FROM customer;                           -- 1-5) 거주지 → 서울이 두 번 (6행)
SELECT DISTINCT city FROM customer;                  -- 1-6) ⭐ DISTINCT = 중복 행 제거 → 서울 한 번 (5행)

SELECT *FROM customer WHERE cno='c101';              -- 2-1) 고객번호 c101 → 홍길동
SELECT * FROM customer WHERE point<=400;             -- 2-2) 400 이하 (400 포함) → 3행
SELECT cname,city,point FROM customer WHERE city='서울' AND point>=500;   -- 2-3) 서울 이면서(AND) 500 이상 → 홍길동
SELECT cname,city,point FROM customer WHERE city='서울' OR point>=500;    -- 2-4) 서울 이거나(OR) 500 이상 → 3행
SELECT cname,city,point FROM customer WHERE point>=350 AND point<=500;    -- 2-5) 350 ~ 500 — 부등호
SELECT cname,city,point FROM customer WHERE point BETWEEN 350 AND 500;    -- 2-5) BETWEEN a AND b = a 이상 b 이하 (양 끝 포함)
SELECT cno,cname,city FROM customer WHERE city='서울' OR city='안양';     -- 2-6) 부등호(=) + OR
SELECT cno,cname,city FROM customer WHERE city IN ('서울','안양');        -- 2-6) ⭐⭐⭐ IN = 목록 중 하나와 같으면
SELECT cno,cname,city FROM customer WHERE city!='서울' AND city!='안양';  -- 2-7) 서울도 아니고 안양도 아닌 (드 모르간)
SELECT cno,cname,city FROM customer WHERE city NOT IN ('서울','안양');    -- 2-7) NOT IN = 목록 어디에도 없으면

SELECT * FROM customer WHERE cname LIKE '정%';       -- 3-1) ⭐⭐ 정 + 뒤에 0글자 이상 → 정씨
SELECT * FROM customer WHERE cname LIKE '%동%';      -- 3-2) 어디든 '동' 이 들어감 → 홍길동 · 신동엽 · 정동우
SELECT * FROM customer WHERE cname LIKE '__우';      -- 3-3) _ 두 개 = 아무 글자 2개 + 세 번째 '우' → 정진우 · 정동우
SELECT * FROM customer WHERE cname LIKE '홍%' OR cname LIKE '박%' OR cname LIKE '정%';             -- 3-4) ⚠️ 정정 — 홍 · 박 · 정씨 (35쪽)
SELECT * FROM customer WHERE cname NOT LIKE '홍%' AND cname NOT LIKE '박%' AND cname NOT LIKE '정%';  -- 3-4) 셋 다 아닌 → NOT LIKE + AND
SELECT cno,cname,point FROM customer WHERE point IS NULL;      -- 3-5) ⭐ 포인트가 없는 (NULL) → 정동우
SELECT cno,cname,point FROM customer WHERE point IS NOT NULL;  -- 3-5) 포인트가 있는 → 5행

SELECT * FROM customer ORDER BY cname ASC;                     -- 4-1) ⭐⭐ 이름 오름차순 (가나다)
SELECT * FROM customer WHERE city='서울' ORDER BY cname ASC;   -- 4-2) 서울만 → 이름 오름차순
SELECT * FROM customer ORDER BY city ASC,point DESC;           -- 4-3) ⭐⭐ 거주지 오름차순 → 같으면 포인트 내림차순
SELECT * FROM customer ORDER BY point DESC, cname ASC, city ASC;   -- 4-4) ⚠️ 빠진 답 — 포인트 ↓ → 이름 ↑ → 거주지 ↑ (42쪽)
SELECT cno, cname, city, point FROM customer ORDER BY 3;       -- 4-5) ORDER BY 3 = SELECT 의 3번째 열(city) 오름차순

4. [실습2] 집계함수 · GROUP BY · HAVING — 성적 · 성적2 (강의자료 45 ~ 52쪽)

✍️ 필기 — COUNT(*) : NULL 을 포함하여 계산 · COUNT(필드) : NULL 을 제외하여 계산 (45쪽 “널값포함 · 널값제외”) · 집계함수 = 여러 개 행의 값들을 계산하여 하나의 결과를 산출하는 함수 (aggregate function)

CREATE TABLE 성적(                                   -- 성적(이름, 점수)
     이름 varchar(9) NOT NULL primary key,           -- 이름 — 기본키
     점수 int                                        -- 점수 — 비워도 됨(NULL)
);                                                   -- CREATE TABLE 끝
INSERT INTO 성적 ( 이름, 점수 ) VALUES ( '홍길동', 87 );   -- 87
INSERT INTO 성적 ( 이름, 점수 ) VALUES ( '임꺽정', 60 );   -- 60
INSERT INTO 성적 ( 이름, 점수 ) VALUES ( '박찬호', 75 );   -- 75
INSERT INTO 성적 ( 이름, 점수 ) VALUES ( '선동열', 70 );   -- 70
INSERT INTO 성적 ( 이름, 점수 ) VALUES ( '홍명보', 90 );   -- 90
INSERT INTO 성적 ( 이름, 점수 ) VALUES ( '차범근', 75 );   -- 75
INSERT INTO 성적 ( 이름, 점수 ) VALUES ( '강성범', 68 );   -- 68
INSERT INTO 성적 ( 이름, 점수 ) VALUES ( '신동엽', null);  -- 응시 안 함 → NULL
CREATE TABLE 성적2 (                                 -- 성적2(이름, 과목, 점수)
    이름 varchar(9) NOT NULL primary key ,           -- 이름 — 기본키
    과목 varchar(8),                                 -- 과목
    점수 int                                         -- 점수
);                                                   -- CREATE TABLE 끝
INSERT INTO 성적2 VALUES ('홍길동', '영어',87 );     -- 영어 87

-- ✍️ 내 답 — 성적(이름, 점수)
SELECT MAX(점수) FROM 성적;                          -- 1-1) 최고 점수
SELECT MIN(점수) FROM 성적;                          -- 1-2) 최저 점수
SELECT SUM(점수) FROM 성적;                          -- 1-3) 합계 — NULL 은 빼고 더함
SELECT AVG(점수) FROM 성적;                          -- 1-4) 평균 — NULL 은 빼고 (7명으로 나눔)
SELECT SUM(점수) / COUNT(점수) FROM 성적;            -- 1-4) 합계 ÷ 점수가 있는 사람 수 = AVG 와 같음
SELECT COUNT(*) FROM 성적;                           -- 1-5) 학생 수 — NULL 행도 셈
SELECT COUNT(점수) FROM 성적;                        -- 1-6) 응시한 학생 수 — 점수가 NULL 인 신동엽은 빼고 셈

-- ✍️ 내 답 — 성적2(이름, 과목, 점수)
SELECT COUNT(DISTINCT 과목) FROM 성적2;              -- 2-1) ⭐ 과목 종류 수 — 중복 빼고 셈
SELECT 과목,COUNT(*) FROM 성적2 GROUP BY 과목;       -- 2-2) ⭐⭐⭐ 과목별 수강생 (⚠️ 교수님 답은 COUNT(점수))
SELECT 과목,ROUND(AVG(점수),1) FROM 성적2 GROUP BY 과목;   -- 2-3) 과목별 평균 — ROUND(값, 1) = 소수 첫째 자리까지 반올림
SELECT 과목,AVG(점수) FROM 성적2 GROUP BY 과목 HAVING AVG(점수)>=75;   -- 2-4) 그룹 조건은 HAVING
SELECT 과목,AVG(점수)                                -- 2-5) ⑤ 과목 · 평균 보여 주기
FROM 성적2                                           -- ① 성적2 에서
WHERE 점수>=70                                       -- ② 70점 이상인 행만 남기고
GROUP BY 과목                                        -- ③ 과목별로 묶어
HAVING AVG(점수)>=75                                 -- ④ 평균 75 이상인 그룹만
ORDER BY AVG(점수) DESC;                             -- ⑥ 평균이 높은 순

⚠️ 팩트체크 — 2-4 문제는 “75 보다 높은” 인데 교수님 답(52쪽)도 내 답도 >= 75 (75 이상) → 국어(75.0000)까지 나와요. 글자 그대로 > 75 면 영어만 나와요 (MySQL 로 확인) → 시험은 교수님 답(>=)대로. 🤖 2-5 에서 수학은 WHERE 가 60 · 68 을 먼저 빼서 (90 + 75) ÷ 2 = 82.5 가 돼 1등 — WHERE(행) 와 HAVING(그룹) 의 순서 차이. ROUND(AVG(점수)) 처럼 자릿수를 빼면 정수로 반올림 → 수학 73 · 국어 75 · 영어 79. 🤖 GROUP BY 결과의 행 순서는 MySQL 이 정해요 (ORDER BY 를 써야 순서가 보장됨).

5. [실습3] tradeDB — CSV 데이터 가져오기 (강의자료 53 ~ 68쪽 · 수업중 제출)

drop database if exists tradeDB;                     -- 같은 이름의 DB 가 있으면 지움 (없으면 그냥 넘어감)
CREATE DATABASE tradeDB                              -- tradeDB 데이터베이스 만들기
default character set utf8mb4;                       -- 한글 · 이모지까지 저장되는 문자 집합
USE tradeDB;                                         -- 이제부터 tradeDB 를 사용
drop table if exists 고객;                           -- 고객 테이블이 있으면 지움
CREATE TABLE 고객 (                                  -- 고객 테이블 만들기 시작
      고객번호  CHAR(5)           PRIMARY KEY ,      -- 고객번호 — 5글자 · 기본키
      고객회사명 VARCHAR(30),                         -- 회사 이름
      담당자명    VARCHAR(20),                        -- 담당자 이름
      담당자직위 VARCHAR(20),                         -- 담당자 직위
      주소          VARCHAR(50),                      -- 주소
      도시          VARCHAR(20),                      -- 도시
      지역          VARCHAR(20),                      -- 지역
      전화번호    VARCHAR(20),                        -- 전화번호
      마일리지    INT                                 -- 마일리지 — 정수
);                                                   -- CREATE TABLE 끝
-- 데이터 넣기: Workbench 에서 고객 테이블 오른쪽 클릭 → Table Data Import Wizard → Browse → 고객.csv → Next → Next → Finish
SELECT COUNT(*) as '몇개임?' FROM 고객;              -- ✍️ 가져온 행 수 확인 → 93
SELECT 고객번호, 담당자명, 마일리지                   -- 1-3) 고객번호 · 담당자명 · 마일리지를
FROM 고객                                            -- 고객 테이블에서
WHERE 마일리지>=100000;                              -- 마일리지 100,000 이상만 (숫자에는 쉼표를 쓰지 않음)
