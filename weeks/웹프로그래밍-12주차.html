3. 이벤트 핸들러 등록 3가지 · window.onload (1117_01_event.html)

✍️ 필기 (수업 파일 주석) — “myButton 처럼 addEventListener 이 많이 쓰고 활용하는 것을 가장 추천함 · 방식: 버블링” · 3번 버튼 onclick 속성 안에 “세미콜론 있어도 없어도 문제없음”

<!DOCTYPE html>                                                     <!-- HTML5 문서 -->
<html lang="ko">                                                    <!-- 문서 시작 · 한국어 -->
<head>                                                              <!-- 문서 정보 -->
    <meta charset="UTF-8">                                          <!-- 한글이 안 깨지게 -->
    <meta name="viewport" content="width=device-width, initial-scale=1.0">   <!-- 폰 화면 폭에 맞춤 -->
    <title>1117_01_event.html</title>                               <!-- 탭 제목 -->
    <script>                                                        // head 의 script — 이때는 아직 body(img · button)가 안 만들어짐
        window.onload = () => {                                     // load 이벤트 핸들러 — 페이지를 이미지까지 다 불러온 뒤에 실행
            const image = document.getElementById('image');         // 이제는 body 의 <img id="image"> 를 찾을 수 있음
            image.src = 'https://www.dongyang.ac.kr/sites/dmu/images/common/logo_b.svg';   // 학교 로고 주소 → 빈 그림에 로고가 뜸
            image.width = 200;                                      // 폭 200px
            image.height = 200;                                     // 높이 200px
        };                                                          // onload 에 함수를 대입한 문장 끝 → ;
    </script>                                                       <!-- head script 끝 -->
</head>                                                             <!-- 문서 정보 끝 -->
<body>                                                              <!-- 화면 -->
    <img id="image"><br>                                            <!-- 처음엔 src 없는 빈 그림 → onload 에서 채움 -->
    <button id="myButton">1번 이벤트핸들러 실행</button>                <!-- ① addEventListener 로 연결 -->
    <button id="myButton2">2번 이벤트핸들러 실행</button>               <!-- ② onclick 프로퍼티로 연결 -->
    <button onclick="btnClick3()">3번 이벤트핸들러 실행</button>         <!-- ③ HTML 속성에 실행할 코드를 바로 씀 -->
    <script>                                                        // body 끝의 script — 버튼들이 이미 만들어진 뒤라 바로 찾음
        const button = document.querySelector("#myButton");         // ① 1번 버튼
        button.addEventListener('click', function () {               // ① addEventListener('이벤트 이름', 핸들러) — ✍️ 가장 추천
            alert("addEventListener() 메서드 방식의 핸들러!");          // 누르면 알림창
        });                                                         // 리스너 등록 끝
        const button2 = document.querySelector("#myButton2");        // ② 2번 버튼
        button2.onclick = function () {                              // ② DOM 프로퍼티 — 요소의 onclick 에 함수를 대입
            alert("DOM 프로퍼티 방식의 핸들러!");                      // 누르면 알림창
        };                                                          // 대입문 끝 → ;
        function btnClick3() {                                      // ③ HTML 속성 onclick="btnClick3()" 이 부르는 함수
            alert('HTML 속성 방식의 핸들러!');                         // 누르면 알림창
        }                                                           // btnClick3 끝
    </script>                                                       <!-- script 끝 -->
</body>                                                             <!-- 화면 끝 -->
</html>                                                             <!-- 문서 끝 -->
▶ 크롬(앱 브라우저)에서 직접 열고 눌러 본 결과 — 페이지가 다 열리면 학교 로고가 200 × 200 으로 뜸 · 1번 → 알림 addEventListener() 메서드 방식의 핸들러! · 2번 → DOM 프로퍼티 방식의 핸들러! · 3번 → HTML 속성 방식의 핸들러!
방식쓰는 법같은 이벤트에 핸들러 여러 개특징 (확인)
① addEventListener (메서드)button.addEventListener('click', 함수)됨 — 등록한 순서대로 모두 실행✍️ 가장 추천 · 이벤트 이름에 on 을 안 붙임 ('onclick' 이라고 쓰면 눌러도 실행 안 됨)
② DOM 프로퍼티button2.onclick = 함수안 됨 — 다시 대입하면 앞의 것을 덮어씀간단 · 함수 이름만 대입 (괄호 X)
③ HTML 속성<button onclick="btnClick3()">속성은 하나HTML 안에 JS 코드를 씀 · 괄호 () 를 붙여야 실행 (onclick="btnClick3" 은 아무 일도 안 일어남)
  • ✍️ “세미콜론 있어도 없어도 문제없음” — 맞아요: onclick="btnClick3();" · onclick="btnClick3()" 둘 다 알림이 떠요 (jsdom 확인)
  • ⚠️ 팩트체크 “방식: 버블링” — addEventListener 는 기본이 버블링 단계(누른 요소 → 부모 쪽으로 올라가며 실행)이고, 세 번째 값에 true 를 주면 캡처링 단계(부모 → 누른 요소)로 바뀌어요. ② ③ 방식도 버블링 단계에서 실행돼요 → 그래서 진짜 차이는 “여러 개를 달 수 있느냐” 예요 (jsdom 확인 · 아래 표)
버튼을 눌렀을 때 실행 순서 (jsdom 확인)등록한 것
1부모 div — addEventListener('click', …, true) (캡처링)
2 · 3버튼 — addEventListener 2개 (둘 다 실행)
4버튼 — onclick 을 두 번 대입 → 두 번째만 실행 (첫째는 덮어써짐)
5부모 div — addEventListener('click', …) (기본 = 버블링 → 버튼 다음)

window.onload — 왜 필요할까

상황결과 (확인)
head 의 script 에서 onload 없이 document.getElementById('image')아직 img 가 안 만들어져서 null → image.width = 200 에서 TypeError: Cannot set properties of null (setting 'width') (jsdom 확인)
window.onload = () => { … } 안에서페이지를 다 불러온 뒤 실행 → img 를 찾음 → 로고가 200 × 200 (크롬 확인)
🤖 다른 방법script 를 body 맨 끝에 두기 — 오늘 버튼 코드처럼 (그때는 버튼이 이미 있음)

4. fetch — 서버에 요청 보내기 (1117_02_fetch.html)

✍️ 필기 (수업 파일 주석) — “뜻: users 의 id 가 1번인 사람을 불러와 주세요 · 2번째 파라미터 생략 가능” · “.com 까지 URI 전체가 URL”

<!DOCTYPE html>                                                     <!-- HTML5 문서 -->
<html lang="ko">                                                    <!-- 문서 시작 -->
<head>                                                              <!-- 문서 정보 -->
    <meta charset="UTF-8">                                          <!-- 한글 -->
    <meta name="viewport" content="width=device-width, initial-scale=1.0">   <!-- 폰 폭 -->
    <title>1117_02_fetch.html</title>                               <!-- 탭 제목 -->
    <script>                                                        // 페이지를 열자마자 실행
        fetch('https://jsonplaceholder.typicode.com/users/1');      // 이 주소로 요청을 보냄 → users 중 id 가 1번인 사람 · 2번째 파라미터(옵션) 생략 → GET
    </script>                                                       <!-- script 끝 -->
</head>                                                             <!-- 문서 정보 끝 -->
<body>                                                              <!-- 화면 (비어 있음) -->
</body>                                                             <!-- 화면 끝 -->
</html>                                                             <!-- 문서 끝 -->
▶ 화면에는 아무것도 안 나와요 — 요청만 보내고 응답을 받아 쓰는 코드는 아직 없어요. 🤖 응답이 왔는지는 크롬 개발자 도구(F12) → Network 탭에서 이름이 1 인 요청을 누르면 보여요 (상태 200 · 응답 JSON).

🤖 같은 주소에 curl 로 직접 요청해서 받은 실제 응답 (헤더 · 바디 일부)

HTTP/1.1 200 OK
Content-Type: application/json; charset=utf-8
Content-Length: 509
Connection: keep-alive

{
  "id": 1,
  "name": "Leanne Graham",
  "username": "Bret",
  …
}
주소 부분https://jsonplaceholder.typicode.com/users/1 에서뜻
프로토콜https://🤖 HTTP 규칙으로 주고받음 (s = 암호화)
도메인 (호스트)jsonplaceholder.typicode.com서버 이름 → DNS 가 IP 로 바꿈 (6번)
경로/users/1서버 안에서 무엇을 — users 중 1번
  • ✍️ “2번째 파라미터 생략 가능” — 맞아요: fetch(주소, 옵션) 의 옵션을 빼면 요청 방식은 GET (node: new Request(주소).method → GET · { method: 'POST' } 를 주면 → POST)
  • ⚠️ 팩트체크 “.com 까지 URI 전체가 URL” — 이 말은 두 가지로 쓰여요. ① 표준(RFC 3986): URI = 자원을 가리키는 이름 전부 · URL = 그중 위치(주소) 로 가리키는 것 → https://jsonplaceholder.typicode.com/users/1 전체가 URL 이자 URI ② HTTP 요청 시작 줄 GET /users/1 HTTP/1.1 에서는 .com 뒤의 /users/1 을 요청 URI(Request-URI · RFC 2616)라고 불러요 → 필기 “.com 까지 · 뒤는 URI · 전체가 URL” 은 ② 뜻으로 보면 맞아요 (교수님이 어느 뜻으로 말했는지는 확인 못 함) · 🤖 .com 까지(https://jsonplaceholder.typicode.com)는 출처(origin) = 프로토콜 + 도메인 (node new URL(…).origin 확인)

5. 🔥 HTTP 메시지 구조 (시험문제)

✍️ 필기 — 구글 검색 “http메시지 구조가 뭐야” 캡처 + “구글 검색 캡처한거 claude 가 그림 그려서 정리하고 사진은 지워도 됨 (시험문제)”

🎬 🔥 HTTP 메시지 그림 — 새 창으로 크게 보기

  • HTTP 메시지 = 클라이언트(웹 브라우저 등)와 서버가 데이터를 주고받을 때 쓰는 정해진 형식의 데이터 단위
  • 종류 2가지 — 요청(Request) 메시지 (클라이언트 → 서버) · 응답(Response) 메시지 (서버 → 클라이언트)
  • 구조 4부분 — 위에서부터 순서대로 시작 줄 → 헤더 → 빈 줄 → 바디
순서부분하는 일요청 예 (fetch users/1)응답 예
①시작 줄 (Start Line)메시지의 목적 — 요청은 방식 · 주소 · 버전 · 응답은 버전 · 상태 코드 · 상태 글GET /users/1 HTTP/1.1HTTP/1.1 200 OK
②헤더 (Header)메시지에 대한 부가 정보(메타데이터) · 이름: 값 줄 여러 개Host: jsonplaceholder.typicode.comContent-Type: application/json; charset=utf-8
③빈 줄 (Empty Line · CRLF)헤더가 끝났다는 표시 → 헤더와 바디를 구분(빈 줄)(빈 줄)
④바디 (Message Body)실제로 보내거나 돌려받는 데이터 (payload) · 선택 사항 — 없을 수도 있음없음 (GET){"id": 1, "name": "Leanne Graham", …}
  • ⚠️ 팩트체크 — 캡처(구글 AI 개요 · 출처 MDN) 내용은 MDN “HTTP messages” 와 같아요 (시작 줄은 한 줄 · 헤더와 바디는 선택 · 빈 줄로 헤더 끝을 알림) · 표의 요청 · 응답 줄은 curl 로 jsonplaceholder 에 직접 요청해서 받은 실제 값이에요 (헤더는 일부만)
  • 🤖 CRLF = 줄 끝 표시 두 글자 \r (Carriage Return) + \n (Line Feed) — 빈 줄 = CRLF 만 있는 줄 · 🤖 응답 시작 줄의 200 = 상태 코드(성공 · OK)
  • 🤖 요청 방식(GET · POST …)은 요청 메시지 시작 줄의 맨 앞에 들어가요 (7번)

6. ✍️ DNS — 도메인 이름을 IP 주소로

✍️ 필기 — “url 주소를 입력하고 ip 주소로 자동으로 바꿔서 웹사이트를 들어가게 해 주는 게 Domain Name System”

  • DNS (Domain Name System) — 사람이 외우기 쉬운 도메인 이름(naver.com)을 컴퓨터가 쓰는 IP 주소로 바꿔 주는 시스템 → 1주차 “웹서비스 프로세스” 슬라이드의 http://210.89.160.88/news.html 혹은 http://naver.com/news.html 이 같은 서버로 가는 이유
  • ⚠️ 팩트체크 — 정확히는 URL 전체가 아니라 그 안의 도메인 이름만 IP 로 바꿔요 (/users/1 같은 경로는 그대로 서버에 보냄)
순서 (🤖)하는 일
1주소창에 https://jsonplaceholder.typicode.com/users/1 입력 (또는 fetch)
2DNS 에 jsonplaceholder.typicode.com 의 IP 를 물어봄 → IP 주소를 받음
3그 IP 의 서버에 요청 메시지 GET /users/1 HTTP/1.1 을 보냄 (5번)
4서버가 응답 메시지 HTTP/1.1 200 OK + JSON 바디를 돌려줌 → 브라우저가 받음

7. ✍️ 요청 방식 · CRUD

✍️ 필기 — “요청방식: get, post, put, delete” · “CRUD (CREATE, READ, UPDATE, DELETE)”

요청 방식 (HTTP 메서드)하는 일 🤖CRUD 짝 🤖fetch 로 🤖
GET가져오기 (조회)Readfetch(주소) — 2번째 파라미터를 빼면 GET
POST새로 만들기 (보내서 저장)Createfetch(주소, { method: 'POST' })
PUT고치기 (통째로 바꿈)Update{ method: 'PUT' }
DELETE지우기Delete{ method: 'DELETE' }
  • CRUD = Create(생성) · Read(조회) · Update(수정) · Delete(삭제) — 데이터를 다루는 기본 4가지 (10주차 필기 “CRUD 가 무엇인가” 와 같음)
  • ⚠️ 관례 · 필수 아님 — GET ↔ Read 처럼 짝짓는 건 흔한 약속(REST 방식)이에요 · 서버가 꼭 이렇게 만들어야 하는 규칙은 아니에요 (10주차 그림 ⑦ 과 같은 표)

8. ✍️ 단일 스레드 · 멀티 스레드 (6주차 복습)

✍️ 필기 — “멀티스레드 단일스레드 차이” (⚠️ 필기 오타 멀티스레스 · 단일스레스 → 스레드)

구분단일 스레드 (single thread)멀티 스레드 (multi thread)
뜻일하는 흐름(스레드)이 1개 → 한 번에 한 가지 일스레드가 여러 개 → 여러 일을 동시에
예자바스크립트 (브라우저 · Node.js 의 JS 실행)🤖 스레드를 여러 개 만드는 프로그램 (예: 자바의 Thread)
장점구조가 단순 · 같은 데이터를 동시에 고치는 충돌이 없음여러 작업을 동시에 처리
단점오래 걸리는 일을 붙잡고 기다리면 뒤의 일이 모두 멈춤 → 그래서 JS 는 비동기스레드끼리 데이터가 꼬이지 않게 관리하기 어려움
  • 🤖 오늘 fetch 와 연결 — 서버 응답은 언제 올지 몰라요 → 단일 스레드인 JS 는 응답을 기다리지 않고 다음 코드를 먼저 실행하고, 응답이 오면 그때 처리해요 (6주차 setTimeout 과 같은 비동기)
// 단일 스레드 — 기다리는 일(타이머 · 서버 응답)은 맡겨 두고 다음 줄을 먼저 실행 (6주차 setTimeout)
console.log("1. 요청 보냄");                    // 첫 번째로 찍힘
setTimeout(() => {                             // 1초 뒤에 실행할 함수를 맡겨 둠 (응답이 1초 뒤에 온다고 생각)
    console.log("3. 응답 도착 → 처리");         // 1초 뒤 → 세 번째로 찍힘
}, 1000);                                      // 1000ms = 1초 · 기다리지 않고 바로 다음 줄로
console.log("2. 기다리지 않고 다음 코드");       // 두 번째로 찍힘
1. 요청 보냄
2. 기다리지 않고 다음 코드
3. 응답 도착 → 처리
