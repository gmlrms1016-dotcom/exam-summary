int main() {                 // 프로그램 시작
    int a = 10;              // a 칸에 10
    int* p = &a;             // p = a 의 주소 → *p 는 a 와 같은 칸
    printf("%d %d", a, *p);  // a 10 · *p = a 의 값 10 → 10 10
    printf("\n");            // 줄바꿈
    *p = 20;                 // p 가 가리키는 칸(a)에 20 → a 도 20
    printf("%d %d", a, *p);  // 20 20
    printf("\n");            // 줄바꿈
    a = 30;                  // a 를 직접 30 으로 → *p 도 30
    printf("%d %d", a, *p);  // a 를 바꿨어도 같은 칸 → 30 30
    return 0;                // 정상 종료
}                            // main 끝

5단계 · 함수 — 값 전달 vs 주소 전달

#include <stdio.h>            // 표준 입출력 헤더

void change(int x, int* y) {  // x = 값 복사본 · y = 주소
    x = x + 10;               // 복사본 x 만 11 → main 의 a 는 1 그대로
    *y = *y + 10;             // y 가 가리키는 b 에 2 + 10 = 12
}                             // change 끝

void swap(int* x, int* y) {   // 두 변수의 주소를 받음
    int t = *x;               // t = a 의 값 1
    *x = *y;                  // a = b 의 값 12
    *y = t;                   // b = t 1
}                             // swap 끝

int main() {                  // 프로그램 시작
    int a = 1, b = 2;         // a = 1 · b = 2
    change(a, &b);            // a 는 값으로 · b 는 주소로 넘김 → a 1 · b 12
    printf("%d %d", a, b);    // 1 12
    printf("\n");             // 줄바꿈
    swap(&a, &b);             // 주소로 넘겨 a ↔ b → a 12 · b 1
    printf("%d %d", a, b);    // 12 1
    return 0;                 // 정상 종료
}                             // main 끝
실행 결과 · 풀이

12 1
  • 💡 change(a, &b) — x 는 a 의 값 복사본이라 a 는 1 그대로 · y 는 b 의 주소라 *y + 10 → b = 12 · swap(&a, &b) — 주소를 넘겨서 a ↔ b → 12 1

#include <stdio.h>                 // 표준 입출력 헤더

void twice(int v[], int n) {       // v = 배열 시작 주소 (9쪽처럼 int v[])
    for (int i = 0; i < n; i++) {  // i = 0, 1, 2

        printf("%d ", *(v + i));   // *(v + i) = v[i] → 6 10 14 (뒤에 공백)

}                                  // show 끝

    show(arr, 3);                  // 6 10 14
    return 0;                      // 정상 종료
}                                  // main 끝
