2. C 프로그램 실행 과정1주차

C 프로그램은 4단계를 거쳐 실행됩니다. 시험에 자주 나오는 핵심 흐름입니다.

단계내용
① 소스코드 작성C 문법으로 코드 작성 → .c 파일
② 컴파일(Compile)컴파일러가 기계어로 변환 → .obj (또는 .o) 오브젝트 파일
③ 링킹(Linking)여러 오브젝트 파일 + 라이브러리를 연결 → .exe (또는 a.out)
④ 실행(Execution)컴파일된 프로그램을 실행해 결과 확인

첫 프로그램 (hello.c) — 한 줄씩 주석

// hello.c — ① 소스코드 작성 (.c 파일)
#include <stdio.h>            // 전처리기: printf 가 들어 있는 stdio.h 를 가져온다
int main(void) {              // 프로그램은 main() 에서 자동으로 시작 · (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    printf("hello world!");   // 화면에 출력 (\n 이 없어서 줄은 안 바뀜)
    return 0;                 // 0 을 돌려주며 프로그램 정상 종료
}   // main 함수 끝
// ② 컴파일 Ctrl+Shift+B → hello.obj   ③ 링크 → hello.exe   ④ 실행 Ctrl+F5

5. 함수와 프로그램 구조2주차

암기 — ① C언어는 함수들로 구성된다 ② C언어는 main() 함수로 시작된다(자동 실행) ③ 다른 함수는 호출해야 실행된다

함수는 표준 라이브러리 함수(printf 등)와 사용자 정의 함수로 나뉩니다.

#include <stdio.h>                  // printf · scanf 를 쓰기 위한 표준 입출력 헤더
void func1(void);            // 함수 원형(선언)

int main(void) {   // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    printf("main() start!!!\n");    // ① main() start!!!
    func1();                 // ← 호출해야 실행된다
    printf("main() end!!!\n");      // ③ func1 에서 돌아와서 → main() end!!!
    return 0;                       // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                   // main 함수 끝
void func1(void) {           // 함수 정의
    printf("func1() start!!!\n");   // ② func1() start!!!
    printf("func1() end!!!\n");     // ② func1() end!!! → main 으로 돌아감
}                                   // func1 함수 끝 → 호출한 곳으로 돌아감

// printf 서식 지정자 — % 뒤의 글자가 "어떤 모양으로 찍을지"를 정한다
#include <stdio.h>    // printf · scanf 를 쓰기 위한 표준 입출력 헤더
int main() {   // () = 괄호를 비워 둔 옛 방식 · int main(void) 는 '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int data = 100;   // 출력해 볼 정수 100
    printf("10진수 : %d\n", data);        // data = 100 → %d 10진수 그대로 → 100
    printf("8진수  : %o\n", data);        // %o  → 144   (100=4+32+64=1100100=001 100 100=144)
    printf("16진수 : %x\n", data);        // %x  → 64    (100=4+32+64=1100100=0110 0100=64)
    printf("16진수 : %X\n", 255);         // 255=1+2+4+8+16+32+64+128=11111111=1111 1111=FF → %X 대문자 16진수 (소문자 %x 는 ff)
    printf("문자   : %c\n", 'A');         // 'A' → %c 문자 1개 → A
    printf("문자열 : %s\n", "안녕하세요");  // "안녕하세요" → %s 문자열 → 안녕하세요
    printf("실수   : %f\n", 3.14);        // 3.14 → %f 소수점 아래 6자리 → 3.140000
    printf("double : %lf\n", 165.7);     // 165.7 → %lf 도 소수점 아래 6자리 → 165.700000
    printf("long   : %ld\n", 123456L);   // 123456L(long) → %ld → 123456

17. 산술 · 대입(복합 대입) 연산자4주차

예제 ① ex01_00-op1.c — 산술 연산자 (a=21, b=10)

// ex01_00-op1.c ① 산술 연산자 + - * / %  (a=21, b=10)
#include <stdio.h>   // printf · scanf 를 쓰기 위한 표준 입출력 헤더
int main() {   // () = 괄호를 비워 둔 옛 방식 · int main(void) 는 '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int a = 21;      // 피연산자 a = 21
    int b = 10;      // 피연산자 b = 10
    int c;           // 결과를 담을 c

    c = a + b;       // 덧셈
    printf("a+b= %d \n", c);     // c = 21 + 10 → 31
    c = a - b;       // 뺄셈
    printf("a-b= %d \n", c);     // c = 21 - 10 → 11
    c = a * b;       // 곱셈
    printf("a*b= %d \n", c);     // c = 21 * 10 → 210
    c = a / b;       // 나눗셈 (정수끼리 → 몫)

// ex01_00-op1.c ② 증감 연산자 ++ --  ★시험에 많이 나옴
#include <stdio.h>                        // printf · scanf 를 쓰기 위한 표준 입출력 헤더
int main() {   // () = 괄호를 비워 둔 옛 방식 · int main(void) 는 '매개변수 없음'을 확실히 적음 — 실행 결과는 같음
    int c;                                // 결과를 담을 c
    int d = 10, e = 10, f = 10, g = 10;   // d · e · f · g 모두 10 으로 시작 (각 증감 연산자 실험용)

    c = d;          // d(10) 을 c 에 대입 → c=10
    d = d + 1;      // d = 10 + 1 → d=11 (d++ 와 같은 뜻)
    printf("c,d: %d, %d \n", c, d);       // c,d: 10, 11

    d = 10;                               // d 를 다시 10 으로
    c = d++;        // 후위: c 에 먼저 대입(10) → 그다음 d 증가(11)
    printf("c,d: %d, %d \n", c, d);       // c,d: 10, 11 (후위 → c 는 증가 전 값)

    c = ++e;        // 전위: e 먼저 증가(11) → 그 값을 c 에 대입(11)
    printf("c,e: %d, %d \n", c, e);       // c,e: 11, 11 (전위 → c 도 증가한 값)

    c = f--;        // 후위: 지금 f(10) 을 c 에 먼저 대입 → c=10 → 그다음 f = 10 - 1 → 9
    printf("c,f: %d, %d \n", c, f);       // c,f: 10, 9

    c = --g;        // 전위: 먼저 g = 10 - 1 → 9 → 그 값을 c 에 대입 → c=9
    printf("c,g: %d, %d \n", c, g);       // c,g: 9, 9
    return 0;                             // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                         // main 함수 끝

// 논리 연산자 ! && || — 결과는 참 1 / 거짓 0 (0 이 아니면 모두 참)

    printf("!0 = %d, !1 = %d\n", !0, !1);                       // NOT: !0 → 거짓의 반대 → 1 · !1 → 참의 반대 → 0 → !0 = 1, !1 = 0
    printf("0&&0=%d 0&&1=%d 1&&0=%d 1&&1=%d\n", 0&&0, 0&&1, 1&&0, 1&&1);   // AND: 둘 다 참일 때만 1 → 0 0 0 1
    printf("0||0=%d 0||1=%d 1||0=%d 1||1=%d\n", 0||0, 0||1, 1||0, 1||1);   // OR: 하나라도 참이면 1 → 0 1 1 1
    printf("0^0=%d 0^1=%d 1^0=%d 1^1=%d\n", 0^0, 0^1, 1^0, 1^1);           // ^ 는 비트 XOR: 서로 다르면 1 → 0 1 1 0
    printf("3 && -1 = %d\n", 3 && -1);                          // 3 → 참 · -1 → 참(0 이 아니면 모두 참) → 참 && 참 → 1
    printf("!5 = %d\n", !5);                                    // 5 는 참 → NOT → 0
    return 0;   // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}               // main 함수 끝

// 09forsum.c — 1 에서 10 까지의 합을 구한다 (for)

    int i, sum;                      // 반복 변수 i 와 합 sum
    for (i = 1, sum = 0; i <= 10; i++) //++i도 가능
        sum += i; // sum = sum + i → 1 + 2 + … + 10 → 55
    printf("1 ~ 10 합: %d\n", sum);   // 1 ~ 10 합: 55

    for (i = 1, sum = 0; i <= 10; )    // 증감식을 비우고
        sum += i++;                    // 후위: 지금 i 를 먼저 더하고 → 그다음 i 증가 → 1 + 2 + … + 10 → 55
    printf("1 ~ 10 합: %d\n", sum);   // 1 ~ 10 합: 55

    for (i = 0, sum = 0; i <= 9; )     // 0 부터 시작하면
        sum += ++i;                    // 전위: i 를 먼저 1 올리고 → 그 값을 더함 → i = 0 에서 1, 2, …, 10 → 55
    printf("1 ~ 10 합: %d\n", sum);   // 1 ~ 10 합: 55

    return 0;                        // 0(정상 종료)을 운영체제에 돌려주고 끝냄

// lab1intpower.c — 표준입력으로 받은 두 양의 정수로 거듭제곱 구하는 함수 intpow() 작성
#define _CRT_SECURE_NO_WARNINGS                         // Visual Studio 에서 scanf 보안 경고(C4996) 끄기 — 맨 위에 써야 함
#include <stdio.h>                                      // printf · scanf 를 쓰기 위한 표준 입출력 헤더
int intpow(int m, int n);                               // 원형 — m 을 n 번 곱한 값을 돌려줌
int main(void) {   // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int m, n;                                           // 밑 m, 지수 n
    printf("정수 m을 n번 제곱합니다.\n");               // 안내
    printf("정수 m n을 계속 입력 >>> ");                // 안내 (\n 없음)
    scanf("%d %d", &m, &n);                             // 두 정수 입력 (예: 2 5)
    printf("\n%d의 %d 제곱은 %d입니다.\n", m, n, intpow(m, n));   // 함수호출 결과를 바로 출력 → 2의 5 제곱은 32입니다.
    return 0;                                           // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                       // main 함수 끝
int intpow(int m, int n) {                              // 거듭제곱 함수
    int mult = 1, i = 1;                                // 곱을 쌓을 mult 는 1 에서 시작
    for (i = 1; i <= n; i++) {                          // n 번 반복
        mult *= m;                                      // mult = mult * m → 1→2→4→8→16→32
    }                                                   // for 끝 → 다음 i
    return mult;                                        // 32 를 돌려줌
}                                                       // intpow 끝

// lab3numguess.c — 1 ~ 100 사이의 난수 알아 맞히기
#define _CRT_SECURE_NO_WARNINGS                         // Visual Studio 에서 scanf 보안 경고(C4996) 끄기 — 맨 위에 써야 함
#include <stdio.h>                                      // printf · scanf · puts 를 쓰기 위한 표준 입출력 헤더
#include <stdlib.h>                                     // rand() · srand() 를 위한 헤더파일
#include <time.h>                                       // time() 을 위한 헤더파일
#define MAX 100                                         // 매크로 상수 MAX = 100
int main(void) {   // (void) = '매개변수 없음'을 확실히 적음 · int main() 은 괄호를 비워 둔 옛 방식 — 실행 결과는 같음
    int guess, input;                                   // 정답 · 입력
    srand((long)time(NULL));                            // 실행마다 다른 난수
    guess = rand() % MAX + 1;                           // 1 ~ 100 중 하나
    printf("1에서 %d 사이에서 한 정수가 결정되었습니다.\n", MAX);   // 안내
    printf("이 정수는 무엇일까요? 입력해 보세요. : ");   // 안내 (\n 없음)
    while (scanf("%d", &input)) {                       // 정수를 읽으면 1(참) → 계속 반복
        if (input > guess)                              // 입력이 크면
            printf("입력한 수 %d보다 작습니다. 다시 입력하세요. : ", input);   // 더 작은 수를 입력하라
        else if (input < guess)                         // 입력이 작으면
            printf("입력한 수 %d보다 큼니다. 다시 입력하세요. : ", input);     // 더 큰 수를 입력하라 (PDF 오타 '큼니다' 그대로)
        else {                                          // 같으면
            puts("정답입니다.");                        // 정답 + 줄바꿈
            break;                                      // while 을 빠져나감
        }                                               // else 끝
    }                                                   // while 끝
    return 0;                                           // 0(정상 종료)을 운영체제에 돌려주고 끝냄
}                                                       // main 함수 끝
