2. C 프로그램 실행 과정1주차

C 프로그램은 4단계를 거쳐 실행됩니다. 시험에 자주 나오는 핵심 흐름입니다.

단계내용
① 소스코드 작성C 문법으로 코드 작성 → .c 파일
② 컴파일(Compile)컴파일러가 기계어로 변환 → .obj (또는 .o) 오브젝트 파일
③ 링킹(Linking)여러 오브젝트 파일 + 라이브러리를 연결 → .exe (또는 a.out)
④ 실행(Execution)컴파일된 프로그램을 실행해 결과 확인

첫 프로그램 (hello.c) — 한 줄씩 주석

// hello.c — ① 소스코드 작성 (.c 파일)
#include <stdio.h>            // 전처리기: printf 가 들어 있는 stdio.h 를 가져온다
int main(void) {              // 프로그램은 main() 에서 자동으로 시작
    printf("hello world!");   // 화면에 출력 (\n 이 없어서 줄은 안 바뀜)
    return 0;                 // 0 을 돌려주며 프로그램 정상 종료
}
// ② 컴파일 Ctrl+Shift+B → hello.obj   ③ 링크 → hello.exe   ④ 실행 Ctrl+F5
hello world!

5. 함수와 프로그램 구조2주차

암기 — ① C언어는 함수들로 구성된다 ② C언어는 main() 함수로 시작된다(자동 실행) ③ 다른 함수는 호출해야 실행된다

함수는 표준 라이브러리 함수(printf 등)와 사용자 정의 함수로 나뉩니다.

#include <stdio.h>
void func1(void);            // 함수 원형(선언)

int main(void) {
    printf("main() start!!!\n");
    func1();                 // ← 호출해야 실행된다
    printf("main() end!!!\n");
    return 0;
}
void func1(void) {           // 함수 정의
    printf("func1() start!!!\n");
    printf("func1() end!!!\n");
}

문제 ① 실행결과를 쓰시오 — [실습 1-05] ex01_05-sizeof.c강의자료 23p

// ex01_05-sizeof.c
#include <stdio.h>
int main() {
    char gender;
    int age;
    float weight;
    double height;

    printf("변수의 할당크기\n");
    printf("gender: %d, age: %d, weight: %d, height: %d\n\n",
        sizeof(gender), sizeof(age), sizeof(weight), sizeof(height));

    printf("자료형 크기\n");
    printf("char: %d, int: %d, float: %d, double: %d\n\n",
        sizeof(char), sizeof(int), sizeof(float), sizeof(double));

    printf("정수형상수: %d, 실수형상수: %d\n",
        sizeof(10), sizeof(3.14159));
    return 0;
}
💡 정답(실행결과) 보기

자료형 크기

풀이 — sizeof 는 크기(바이트 수)를 정수로 돌려주므로 %d 로 출력합니다. 변수의 크기는 자료형 크기와 같고(char 1 · int 4 · float 4 · double 8), sizeof(10)은 정수 상수라 int → 4, sizeof(3.14159)는 실수 상수라 double → 8입니다.
수업 중 “시험문제!” — sizeof(float) 를 %f, sizeof(double) 을 %lf, sizeof(char) 를 %c 로 출력하는 코드가 나오면 틀린 것 → 전부 %d (결과: int 4 · float 4 · double 8 · char 1)

// sample03_01.c — 변수에 값을 저장하고 출력

예제 — 같은 비트, 다른 값

// 부호 비트 — 1byte = 8bit, 맨 앞 비트가 0 이면 양수 · 1 이면 음수 (signed)
#include <stdio.h>
int main() {
    signed char   s1 = 127;    // 0111 1111 → signed 최댓값
    signed char   s2 = -128;   // 1000 0000 → signed 최솟값
    unsigned char u1 = 0;      // 0000 0000 → unsigned 최솟값
    unsigned char u2 = 255;    // 1111 1111 → unsigned 최댓값
    printf("signed char   : %d ~ %d\n", s2, s1);
    printf("unsigned char : %d ~ %d\n", u1, u2);

    // 같은 비트라도 signed / unsigned 에 따라 값이 달라진다
    unsigned char u3 = 128;    // 1000 0000 → 부호 없음 : 128
    signed char   s3 = 128;    // 1000 0000 → 부호 있음 : 맨 앞 1 = 음수 → -128
    unsigned char u4 = 255;    // 1111 1111 → 부호 없음 : 255
    signed char   s4 = 255;    // 1111 1111 → 부호 있음 : -1
    printf("1000 0000 → unsigned %d / signed %d\n", u3, s3);
    printf("1111 1111 → unsigned %d / signed %d\n", u4, s4);

    // int n;          ← 기본이 signed (signed int n; 과 같다) : 음수·양수 모두
    // unsigned int n; ← 부호 없는 int : 0 이상만 저장
    return 0;
}

// printf 서식 지정자 — % 뒤의 글자가 "어떤 모양으로 찍을지"를 정한다

    printf("long   : %ld\n", 123456L);   // %ld → long 정수

    float value = 2.1f;                  // 강의자료 10p
    printf("float  : %f\n", value);      // 2.100000 (6자리)

예제 — 우선순위 (필기 3<<2+5<<3)

// 연산자 우선순위 — 괄호 > 단항 > 산술 > (시프트) > 관계 > 논리 > 조건 > 대입 > 콤마
#include <stdio.h>
int main() {
    printf("%d\n", 3 << 2 + 5 << 3);        // + 가 << 보다 먼저 → 3 << 7 << 3 = 3072
    printf("%d\n", (3 << 2) + (5 << 3));    // 괄호 먼저 → 12 + 40 = 52

// ex01_00-op1.c ② 증감 연산자 ++ --  ★시험에 많이 나옴
#include <stdio.h>
int main() {
    int c;
    int d = 10, e = 10, f = 10, g = 10;

    c = d;          // c=10
    d = d + 1;      // d=11  (d++ 와 같은 뜻)
    printf("c,d: %d, %d \n", c, d);

    d = 10;
    c = d++;        // 후위: c 에 먼저 대입(10) → 그다음 d 증가(11)
    printf("c,d: %d, %d \n", c, d);

    printf("c,e: %d, %d \n", c, e);

    c = f--;        // 후위: c=10, f=9
    printf("c,f: %d, %d \n", c, f);

    c = --g;        // 전위: g=9, c=9
    printf("c,g: %d, %d \n", c, g);
    return 0;

    int n = 5;
    int r;
    r = (n > 10) && (++n > 0);
    printf("r=%d, n=%d\n", r, n);
    r = (n > 1) || (++n > 0);
