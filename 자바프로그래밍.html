// switch 문 — 학점 매기기
public class 연습 {                              // 클래스 시작 — 파일 이름은 연습.java
    public static void main(String[] args) {   // 프로그램이 시작되는 main 메소드
        int score = 85;                        // 점수 85
        char grade;                            // 학점을 담을 문자 변수 (아래 switch 에서 정해짐)
        switch (score/10) {             // 85 / 10 → 8 (정수 나눗셈)
          case 10:                             // 10 (100점) → 아래 case 9 와 같이 A
          case 9:  grade='A'; break;    // break 없으면 다음 case로 계속(fall-through)
          case 8:  grade='B'; break;           // 8 → B 를 넣고 break 로 switch 탈출
          case 7:  grade='C'; break;           // 7 → C
          case 6:  grade='D'; break;           // 6 → D
          default: grade='F';           // 일치하는 것 없을 때 (생략 가능)
        }                                      // switch 끝
        System.out.println("학점은 " + grade);   // 8 → case 8 → B → 학점은 B
    }                                          // main 끝 → 프로그램 종료
}                                              // 클래스 끝
⚠️ case 값은 문자·정수·문자열 리터럴만 가능합니다. (실수·변수·수식 불가)

        System.out.println("a=" + a + ", b=" + b + ", c=" + c);   // a=6, b=9, c=1
        int d = 3;                             // 증감 연산자 실험용 d = 3
        a = d++;    // 후위: 지금 d(3) 을 a 에 먼저 대입 → a=3 → 그다음 d = 3 + 1 → 4
        System.out.println("a=" + a + ", d=" + d);                // a=3, d=4
        a = ++d;    // 전위: 먼저 d = 4 + 1 → 5 → 그 값을 a 에 대입 → a=5, d=5
        System.out.println("a=" + a + ", d=" + d);                // a=5, d=5
        a = d--;    // 후위: 지금 d(5) 를 a 에 먼저 대입 → a=5 → 그다음 d = 5 - 1 → 4
        System.out.println("a=" + a + ", d=" + d);                // a=5, d=4
        a = --d;    // 전위: 먼저 d = 4 - 1 → 3 → 그 값을 a 에 대입 → a=3, d=3
        System.out.println("a=" + a + ", d=" + d);                // a=3, d=3
    }                                          // main 끝 → 프로그램 종료
}                                              // 클래스 끝

예제 3-5 ContinueExample — 양수만 더하기

// ContinueExample.java (예제 3-5) — continue 로 양수만 더하기
import java.util.Scanner;                      // Scanner(키보드 입력)를 쓰려고 불러옴

public class 연습 {   // 원래 이름 ContinueExample → 파일 이름도 연습.java
    public static void main(String[] args) {   // 프로그램이 시작되는 main 메소드
        Scanner sc = new Scanner(System.in);   // 키보드(System.in)로 입력받는 Scanner 객체 sc 만들기
        System.out.println("정수를 5개 입력하세요.");   // 안내문 출력
        int sum = 0;                           // 양수의 합 0 으로 시작
        for (int i = 0; i < 5; i++) {          // i = 0 → 4 → 5번 반복 (정수 5개 입력)
            int n = sc.nextInt();              // 정수 하나 입력 → n
            if (n <= 0) {                      // 0 이하(0 · 음수)면
                continue;               // 0 이하면 아래를 건너뛰고 다음 반복으로
            } else {                           // if 끝 · 조건이 거짓이면 아래 else 실행
                sum += n;               // 양수만 더한다
            }                                  // else 끝
        }                                      // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        System.out.println("양수의 합은 " + sum);   // 양수만 더한 합 출력 → 예: 5 -2 6 10 -4 → 5+6+10 → 양수의 합은 21
        sc.close();                            // 다 쓴 Scanner 닫기
    }                                          // main 끝 → 프로그램 종료
}                                              // 클래스 끝

        System.out.println("종료합니다...");         // while 을 빠져나온 뒤 → 종료합니다...

    }                                           // main 끝 → 프로그램 종료

// ScoreAverrage.java (예제 3-10) — 2차원 배열로 4년(8학기) 평점 평균
public class 연습 {   // 원래 이름 ScoreAverrage → 파일 이름도 연습.java
    public static void main(String[] args) {    // 프로그램이 시작되는 main 메소드
        double score[][] = {{3.3, 3.4},     // 1학년 1, 2학기
                            {3.5, 3.6},     // 2학년
                            {3.7, 4.0},     // 3학년
                            {4.1, 4.2}};    // 4학년

        double sum = 0;                         // 8학기 평점 합계 0 으로 시작
        for (int year = 0; year < score.length; year++) {             // 학년(행) 4번
            for (int term = 0; term < score[year].length; term++) {   // 학기(열) 2번
                sum += score[year][term];       // sum = sum + 그 학기 평점 → 8번 더하면 29.8
            }                                   // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        }                                       // for 끝 → 증감식 실행 후 조건 다시 검사 (거짓이면 빠져나옴)
        int n = score.length;       // score.length → 행(학년) 개수 → 4
        int m = score[0].length;    // score[0] = {3.3, 3.4} → 길이 → 열(학기) 개수 2
        System.out.println("4년 전체 평점 평균은 " + sum / (n * m));   // 29.8 ÷ (4 × 2) → 29.8 / 8 → 4년 전체 평점 평균은 3.725
    }                                           // main 끝 → 프로그램 종료
}                                               // 클래스 끝
