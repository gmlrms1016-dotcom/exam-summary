6. 키(Key) ★핵심3주차

종류설명
기본키 (Primary Key)각 투플을 유일하게 구별하는 하나 이상의 속성 집합 · 중복 불가 · 필수 입력
외래키 (Foreign Key)다른 테이블을 참조하는 키
복합키두 개 이상의 속성을 묶어 기본키로 사용
-- 복합키 예: 수강(학번, 과목번호, 성적)
CREATE TABLE 수강(     -- 수강 테이블 만들기 시작
  학번      char(5),   -- 학번 — 고정 길이 문자 5자리
  과목번호  char(5),     -- 과목번호 — 고정 길이 문자 5자리
  성적      int,       -- 성적 — 정수
  PRIMARY KEY(학번, 과목번호)   -- 복합키
);                   -- CREATE TABLE 끝

7. MySQL 실습2~3주차

실습 4단계 ★ — ① 데이터베이스(스키마) 생성 → ② 테이블 생성 → ③ 데이터 입력 → ④ 데이터 조회

① 학생 테이블

CREATE DATABASE testdb DEFAULT CHARACTER SET utf8mb4;   -- testdb 데이터베이스 만들기 (한글 · 이모지까지 저장되는 utf8mb4)
USE testdb;                                     -- 이제부터 testdb 를 사용

CREATE TABLE 학생(                                -- 학생 테이블 만들기 시작
  학번      int          NOT NULL,                -- 학번 — 정수 · 비워 둘 수 없음(NOT NULL)
  이름      varchar(30)  NOT NULL,                -- 이름 — 최대 30자 가변 길이 문자 · 필수
  성적      int,                                  -- 성적 — 정수 (비워도 됨)
  지도교수  char(2),                                -- 지도교수 — 문자 2자리
  PRIMARY KEY(학번)                               -- 학번을 기본키로 (중복 X · NULL X)
);                                              -- CREATE TABLE 끝
insert into 학생 values('100','Kim',92,'P1');     -- 한 행 추가 → 100 · Kim · 92 · P1 ('100' 은 int 칸이라 숫자 100 으로 들어감)
insert into 학생 values('200','Park',88,'P2');    -- 한 행 추가 → 200 · Park · 88 · P2
select * from 학생;                               -- 학생 테이블의 모든 열 · 모든 행 조회 → 2행

② 제품 테이블 + 조건 검색

USE testdb;                                     -- testdb 사용
DROP TABLE if exists 제품;                        -- 제품 테이블이 있으면 지움 (다시 실행해도 오류 안 나게)
CREATE TABLE 제품(                                -- 제품 테이블 만들기 시작
  제품번호  char(3)      NOT NULL,                  -- 제품번호 — 문자 3자리 · 필수
  제품명    varchar(20),                           -- 제품명 — 최대 20자
  재고량    int,                                   -- 재고량 — 정수
  단가      int,                                  -- 단가 — 정수
  제조업체  varchar(20),                            -- 제조업체 — 최대 20자
  PRIMARY KEY(제품번호)                             -- 제품번호를 기본키로
);                                              -- CREATE TABLE 끝
INSERT INTO 제품 VALUES('P01','그냥만두',5000,4500,'대한식품');   -- P01 · 그냥만두 · 재고 5000 · 단가 4500 · 대한식품
INSERT INTO 제품 VALUES('P02','매운쫄면',2500,5500,'민국푸드');   -- P02 · 매운쫄면 · 재고 2500 · 단가 5500 · 민국푸드

-- 제품명, 재고량 검색
SELECT 제품명, 재고량 FROM 제품;                        -- 두 열만 → 그냥만두 5000 / 매운쫄면 2500
-- 제조업체가 '대한식품'인 제품명·단가·제조업체 검색
SELECT 제품명, 단가, 제조업체 FROM 제품 WHERE 제조업체='대한식품'; -- 조건에 맞는 행만 → 그냥만두 4500 대한식품

9. 기본키 · 외래키 ★시험4주차

구분내용 (10p · 11p · 39p · 40p)
기본키
PRIMARY KEY		각 투플을 유일하게 구별할 수 있는 하나 이상의 속성 집합
중복 불가 · 필수 입력(NOT NULL)
외래키
FOREIGN KEY		다른 릴레이션의 기본키를 참조하는 키
참조되는 릴레이션의 기본키 값에 반드시 존재해야 한다 → 참조 무결성 제약조건
외래키는 널(NULL) 값 입력을 허용한다
⭐ 시험문제 (필기) — 외래키는 참조하는 열과 참조되는 열의 데이터 형식이 같아야 한다. 예) 학생.학과번호 int ↔ 학과.학과번호 int
🔑 부모 · 자식 테이블 — 기본키를 주는 쪽이 부모(학과), 외래키로 참조하는 쪽이 자식(학생)입니다.

20. 뷰(view) ★핵심5주차

💡 뷰 = 다른 테이블을 기반으로 만든 가상의 테이블. 실제 데이터는 원래 테이블에 있고 뷰는 보여 주는 창입니다.
🎯 장점 2가지 ★ — ① 보안(필요한 열만 보여 줌) ② 사용의 편의성(복잡한 조회를 이름 하나로)
-- 뷰 정의(생성)
CREATE VIEW 뷰이름   -- 뷰(가상 테이블) 만들기 — 이름 정하기
as                -- as 뒤의 SELECT 결과가 뷰의 내용
SELECT 열이름        -- 뷰에 보여 줄 열
FROM   테이블이름;     -- 원래 데이터가 있는 테이블

DROP VIEW 뷰이름;        -- 뷰 삭제
DROP TABLE 테이블이름;   -- 테이블 삭제

as                 -- 아래 SELECT 결과가 뷰가 됨

⭐ 핵심 — 뷰에 넣은 데이터는 기반 테이블(직원)에 저장됩니다. 뷰에 없는 열(주소 · 연봉)은 값을 주지 않았으니 주소 NULL · 연봉 기본값 0이 됩니다.
